-- Sell items that are not perfumes (for example a Tester Kit) on the Sales page
-- Run once in the Supabase SQL editor. Safe to re-run.
--
-- A sale line can now be either a perfume size (as before) or an "other item": a description,
-- a quantity and a price. Other items:
--   * are part of the sale total and are booked to Sales (4000) like the rest of the invoice;
--   * take no stock and book no cost of goods sold;
--   * go through the same cash / bank / credit, approval, invoice and void flow as any sale.
-- post_sale_internal and void_sale need no change: they only touch stock for lines that have a size.
-- Returns and exchanges are for perfume lines only. To undo a sale of an other item, void the sale.

begin;

alter table public.sale_lines alter column size_id drop not null;

alter table public.sale_lines add column if not exists description text;

alter table public.sale_lines drop constraint if exists sale_lines_item_chk;
alter table public.sale_lines add constraint sale_lines_item_chk check (
  (size_id is not null and description is null)
  or (size_id is null and length(trim(description)) > 0)
);

-- record_sale: same arguments as before. A line without size_id but with a description is an "other item".
create or replace function public.record_sale(
  p_date date,
  p_customer uuid,
  p_payment text,
  p_notes text,
  p_lines jsonb,
  p_down_payment numeric default 0,
  p_down_method text default null,
  p_down_reference text default null
)
returns uuid
language plpgsql
security definer
set search_path to 'public'
as $function$
declare
  v_id uuid := gen_random_uuid();
  v_no bigint;
  v_today date := (now() at time zone 'Asia/Manila')::date;
  v_date date := coalesce(p_date, (now() at time zone 'Asia/Manila')::date);
  v_line jsonb;
  v_n int := 0;
  v_qty int;
  v_sz public.perfume_sizes;
  v_pname text;
  v_price numeric(14,2);
  v_total numeric(14,2) := 0;
  v_down numeric(14,2) := round(coalesce(p_down_payment, 0), 2);
  v_dmethod text := nullif(trim(coalesce(p_down_method, '')), '');
  v_dref text := nullif(trim(coalesce(p_down_reference, '')), '');
  v_tier text := 'retail';
  v_desc text;   -- price list: follows the customer's type (retail, reseller or distributor)
begin
  if not public.is_active_user() then raise exception 'Not allowed.'; end if;
  if p_payment is null or p_payment not in ('cash','bank','credit') then
    raise exception 'Choose how the customer pays.';
  end if;
  if p_lines is null or jsonb_typeof(p_lines) <> 'array' or jsonb_array_length(p_lines) < 1 then
    raise exception 'Add at least one item.';
  end if;

  if v_down < 0 then raise exception 'The upfront payment cannot be negative.'; end if;
  if v_down > 0 then
    if p_payment <> 'credit' then
      raise exception 'An upfront payment only applies to credit sales.';
    end if;
    if v_dmethod is null or v_dmethod not in ('cash','bank') then
      raise exception 'Choose whether the upfront payment was cash or bank transfer.';
    end if;
    if v_dmethod = 'bank' and v_dref is null then
      raise exception 'Enter the bank transfer reference number for the upfront payment.';
    end if;
    if v_date > v_today then
      raise exception 'A sale with an upfront payment cannot be dated in the future.';
    end if;
  else
    v_dmethod := null;
    v_dref := null;
  end if;

  if p_customer is not null then
    perform 1 from public.customers where id = p_customer and status = 'Active';
    if not found then raise exception 'Customer not found or inactive.'; end if;
    select lower(coalesce(category, 'Retail')) into v_tier from public.customers where id = p_customer;
  end if;
  if p_payment = 'credit' then
    if p_customer is null then raise exception 'A credit sale needs a customer.'; end if;
    perform 1 from public.customer_credit where customer_id = p_customer and credit_allowed;
    if not found then
      raise exception 'This customer is not allowed credit. Choose cash or bank, or ask the owner to allow credit.';
    end if;
  end if;

  v_no := nextval('public.sale_no_seq');
  insert into public.sales (id, sale_no, sale_date, customer_id, payment_type, status, notes,
                            down_payment, down_method, down_reference, price_tier)
  values (v_id, v_no, v_date, p_customer, p_payment, 'Pending', nullif(trim(coalesce(p_notes, '')), ''),
          v_down, v_dmethod, v_dref, v_tier);

  for v_line in select * from jsonb_array_elements(p_lines) loop
    v_n := v_n + 1;
    v_qty := coalesce((v_line->>'qty')::int, 0);
    if v_qty <= 0 then raise exception 'Line %: quantity must be more than zero.', v_n; end if;

    -- An item that is not a perfume (for example a Tester Kit): a description and a price.
    -- No stock is taken and no cost is booked; it is part of the sale total and is booked as Sales.
    if nullif(v_line->>'size_id', '') is null then
      v_desc := nullif(trim(coalesce(v_line->>'description', '')), '');
      if v_desc is null then raise exception 'Line %: choose a perfume and size, or describe the item.', v_n; end if;
      if length(v_desc) > 80 then raise exception 'Line %: the description is too long (80 characters at most).', v_n; end if;
      v_price := round(coalesce(nullif(v_line->>'unit_price', '')::numeric, 0), 2);
      if v_price <= 0 then raise exception 'Line %: the price must be more than zero.', v_n; end if;
      insert into public.sale_lines (sale_id, line_no, size_id, description, qty, unit_price, list_price, amount)
      values (v_id, v_n, null, v_desc, v_qty, v_price, v_price, round(v_qty * v_price, 2));
      v_total := v_total + round(v_qty * v_price, 2);
      continue;
    end if;

    select ps.* into v_sz
      from public.perfume_sizes ps join public.perfumes pf on pf.id = ps.perfume_id
     where ps.id = nullif(v_line->>'size_id', '')::uuid and ps.is_active and pf.status <> 'Discontinued';
    if not found then raise exception 'Line %: choose a perfume and size.', v_n; end if;
    select name into v_pname from public.perfumes where id = v_sz.perfume_id;
    if v_sz.price <= 0 then
      raise exception 'No selling price is set for % % ml. Set it in the Signature Collection first.', v_pname, trim_scale(v_sz.size_ml);
    end if;

    -- the price list follows the customer's type; a price of 0 on a list means "use the retail price"
    v_price := round(coalesce(nullif(v_line->>'unit_price', '')::numeric,
                              case when v_tier = 'reseller' and v_sz.reseller_price > 0 then v_sz.reseller_price
                                   when v_tier = 'distributor' and v_sz.distributor_price > 0 then v_sz.distributor_price
                                   else v_sz.price end), 2);
    if v_price <= 0 then raise exception 'Line %: the price must be more than zero.', v_n; end if;

    insert into public.sale_lines (sale_id, line_no, size_id, qty, unit_price, list_price, amount)
    values (v_id, v_n, v_sz.id, v_qty, v_price, v_sz.price, round(v_qty * v_price, 2));
    v_total := v_total + round(v_qty * v_price, 2);
  end loop;

  if v_down > 0 and v_down >= v_total then
    raise exception 'The upfront payment must be less than the sale total. For a full payment, use Cash or Bank transfer.';
  end if;

  update public.sales set total = v_total where id = v_id;

  -- cash and bank sales post immediately; credit waits for the owner
  if p_payment <> 'credit' then
    perform public.post_sale_internal(v_id, v_date);
  end if;
  return v_id;
end;
$function$;

revoke all on function public.record_sale(date, uuid, text, text, jsonb, numeric, text, text) from public, anon;
grant execute on function public.record_sale(date, uuid, text, text, jsonb, numeric, text, text) to authenticated;

commit;

-- make the API pick up the changed column and function
notify pgrst, 'reload schema';
