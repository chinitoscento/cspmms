-- Saved stock count reports (Audit page)
-- Run once in the Supabase SQL editor. Safe to re-run.
--
-- A saved report is a frozen copy of a physical count: what the system said, what was
-- counted, and the difference, as of the moment it was saved.
-- It never touches stock, stock layers or the ledger. Changing stock or the books still
-- needs a formal entry (Stock Write-offs, or a correction), exactly as before.
-- Reports cannot be edited or deleted once saved.

begin;

create sequence if not exists public.count_report_no_seq;

-- 1. Tables -------------------------------------------------------------------
create table if not exists public.stock_count_reports (
  id            uuid primary key default gen_random_uuid(),
  report_no     bigint not null unique default nextval('public.count_report_no_seq'),
  count_date    date not null,
  counted_by    text,
  notes         text,
  kinds         text[] not null,
  items_total   integer not null,
  items_counted integer not null,
  items_off     integer not null,
  value_diff    numeric(14,2) not null default 0,
  saved_by      uuid default auth.uid(),
  saved_at      timestamptz not null default now()
);

create table if not exists public.stock_count_report_lines (
  id           bigint generated always as identity primary key,
  report_id    uuid not null references public.stock_count_reports(id) on delete restrict,
  stock_kind   text not null,
  item_id      text not null,
  item_name    text not null,
  group_name   text,
  unit         text,
  system_qty   numeric not null,
  system_value numeric(14,2) not null,
  counted_qty  numeric,            -- null = this item was not counted
  diff_qty     numeric,
  value_diff   numeric(14,2)
);

create index if not exists stock_count_report_lines_report_idx
  on public.stock_count_report_lines (report_id);

-- 2. Read-only for the owner; written only by the function below -----------------
alter table public.stock_count_reports enable row level security;
alter table public.stock_count_report_lines enable row level security;

drop policy if exists stock_count_reports_owner_read on public.stock_count_reports;
create policy stock_count_reports_owner_read on public.stock_count_reports
  for select to authenticated using (public.is_owner());

drop policy if exists stock_count_report_lines_owner_read on public.stock_count_report_lines;
create policy stock_count_report_lines_owner_read on public.stock_count_report_lines
  for select to authenticated using (public.is_owner());

revoke all on public.stock_count_reports from anon, authenticated;
revoke all on public.stock_count_report_lines from anon, authenticated;
grant select on public.stock_count_reports to authenticated;
grant select on public.stock_count_report_lines to authenticated;

-- 3. Saved reports cannot be changed or deleted ----------------------------------
create or replace function public.stock_count_report_frozen()
returns trigger
language plpgsql
as $function$
begin
  raise exception 'A saved stock count report cannot be changed or deleted.';
end;
$function$;

drop trigger if exists stock_count_reports_frozen on public.stock_count_reports;
create trigger stock_count_reports_frozen
  before update or delete on public.stock_count_reports
  for each row execute function public.stock_count_report_frozen();

drop trigger if exists stock_count_report_lines_frozen on public.stock_count_report_lines;
create trigger stock_count_report_lines_frozen
  before update or delete on public.stock_count_report_lines
  for each row execute function public.stock_count_report_frozen();

-- 4. save_stock_count_report ------------------------------------------------------
-- The browser sends only what was counted. The system quantities are read here, at the
-- moment of saving, so the report cannot be edited on the way in.
-- p_lines: [{ "stock_kind": "finished", "item_id": "...", "counted": 12 or null }]
-- Every item of a kind that has at least one counted item is stored; items left blank
-- are saved as "not counted".
create or replace function public.save_stock_count_report(
  p_count_date date,
  p_counted_by text,
  p_notes text,
  p_lines jsonb
)
returns uuid
language plpgsql
security definer
set search_path to 'public'
as $function$
declare
  v_today date := (now() at time zone 'Asia/Manila')::date;
  v_id uuid := gen_random_uuid();
  v_rows jsonb;
  v_total int;
  v_counted int;
  v_off int;
  v_val numeric(14,2);
  v_kinds text[];
begin
  if not public.is_owner() then raise exception 'Only the owner can save a stock count report.'; end if;
  if p_count_date is null then raise exception 'Choose the date of the count.'; end if;
  if p_count_date > v_today then raise exception 'The count date cannot be in the future.'; end if;
  if p_lines is null or jsonb_typeof(p_lines) <> 'array' or jsonb_array_length(p_lines) < 1 then
    raise exception 'Enter at least one counted quantity.';
  end if;
  if exists (
    select 1 from jsonb_array_elements(p_lines) e
    where nullif(e->>'counted', '') is not null and (e->>'counted')::numeric < 0
  ) then
    raise exception 'A counted quantity cannot be negative.';
  end if;

  select coalesce(jsonb_agg(jsonb_build_object(
           'stock_kind', s.stock_kind,
           'item_id', s.item_id::text,
           'item_name', s.item_name,
           'group_name', s.group_name,
           'unit', s.unit,
           'system_qty', s.system_qty,
           'system_value', s.system_value,
           'counted', t.counted)), '[]'::jsonb)
    into v_rows
  from public.audit_count_sheet() s
  left join (
    select e->>'item_id' as item_id, e->>'stock_kind' as stock_kind, nullif(e->>'counted', '')::numeric as counted
    from jsonb_array_elements(p_lines) e
  ) t on t.item_id = s.item_id::text and t.stock_kind = s.stock_kind
  where s.stock_kind in (
    select e->>'stock_kind' from jsonb_array_elements(p_lines) e where nullif(e->>'counted', '') is not null
  );

  select count(*),
         count(*) filter (where (r->>'counted') is not null),
         count(*) filter (where (r->>'counted') is not null
                            and abs((r->>'counted')::numeric - (r->>'system_qty')::numeric) > 0.0005),
         coalesce(sum(case
           when (r->>'counted') is not null and (r->>'system_qty')::numeric > 0
             then round(((r->>'counted')::numeric - (r->>'system_qty')::numeric)
                        * (r->>'system_value')::numeric / (r->>'system_qty')::numeric, 2)
           else 0 end), 0),
         array(select distinct r2->>'stock_kind' from jsonb_array_elements(v_rows) r2 order by 1)
    into v_total, v_counted, v_off, v_val, v_kinds
  from jsonb_array_elements(v_rows) r;

  if v_counted = 0 then raise exception 'Enter at least one counted quantity.'; end if;

  insert into public.stock_count_reports
    (id, count_date, counted_by, notes, kinds, items_total, items_counted, items_off, value_diff)
  values
    (v_id, p_count_date, nullif(trim(coalesce(p_counted_by, '')), ''), nullif(trim(coalesce(p_notes, '')), ''),
     v_kinds, v_total, v_counted, v_off, v_val);

  insert into public.stock_count_report_lines
    (report_id, stock_kind, item_id, item_name, group_name, unit, system_qty, system_value,
     counted_qty, diff_qty, value_diff)
  select v_id, r->>'stock_kind', r->>'item_id', r->>'item_name', r->>'group_name', r->>'unit',
         (r->>'system_qty')::numeric, round((r->>'system_value')::numeric, 2),
         (r->>'counted')::numeric,
         case when (r->>'counted') is not null
              then (r->>'counted')::numeric - (r->>'system_qty')::numeric end,
         case when (r->>'counted') is not null and (r->>'system_qty')::numeric > 0
              then round(((r->>'counted')::numeric - (r->>'system_qty')::numeric)
                         * (r->>'system_value')::numeric / (r->>'system_qty')::numeric, 2)
              when (r->>'counted') is not null then 0 end
  from jsonb_array_elements(v_rows) r;

  return v_id;
end;
$function$;

revoke all on function public.save_stock_count_report(date, text, text, jsonb) from public, anon;
grant execute on function public.save_stock_count_report(date, text, text, jsonb) to authenticated;

commit;

-- make the API pick up the new table and function
notify pgrst, 'reload schema';
