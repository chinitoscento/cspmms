-- =====================================================================
-- Chinito Scento - MARKETING GIFTS: several perfumes in one go.
-- Run in the Supabase SQL Editor AFTER 31_marketing_gifts.sql. Safe to re-run.
--
-- Records one gift per perfume line (each gets its own GF number and can be voided on its own),
-- all inside ONE transaction: if any line fails (for example not enough stock), nothing is saved.
-- Nothing existing is changed.
-- =====================================================================

create or replace function public.record_marketing_gifts(
  p_date date, p_recipient text, p_notes text, p_lines jsonb
) returns integer
language plpgsql security definer set search_path = public as $$
declare
  l jsonb;
  n integer := 0;
begin
  if not public.is_active_user() then raise exception 'Not allowed.'; end if;
  if p_lines is null or jsonb_typeof(p_lines) <> 'array' or jsonb_array_length(p_lines) < 1 then
    raise exception 'Add at least one perfume.';
  end if;

  for l in select * from jsonb_array_elements(p_lines) loop
    perform public.record_marketing_gift(
      p_date, (l->>'size_id')::uuid, (l->>'qty')::integer, p_recipient, p_notes);
    n := n + 1;
  end loop;

  return n;
end;
$$;

revoke all on function public.record_marketing_gifts(date, text, text, jsonb) from public, anon;
grant execute on function public.record_marketing_gifts(date, text, text, jsonb) to authenticated;

notify pgrst, 'reload schema';
