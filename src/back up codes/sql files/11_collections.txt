-- =====================================================================
-- Chinito Scento - Collections: customer payments, receivable balances,
-- customer ledger and statement of account.
-- Run after 09_sales.sql. Safe to re-run.
-- NOTE: 09_sales.sql defines approve_sales_return and void_sale in their older
-- form. If you ever re-run 09, run this file again afterwards.
--
-- DECISIONS
--   * Only the owner records a customer payment (cash or bank transfer).
--   * A payment settles open credit invoices, oldest due date first, or the
--     invoices the owner picks. It can never settle more than an invoice's balance.
--   * An overpayment is NOT kept as customer credit.
--       - Cash: refused. Take only what is owed.
--       - Bank transfer: the owner may record the full amount received and tick
--         "refund the excess by bank transfer". The same journal entry then
--         shows the money coming in AND going back out, so the audit trail is complete.
--
-- BOOKS
--   Payment received:   Dr 1010 Cash | 1020 Bank      amount received
--                         Cr 1100 Accounts Receivable   applied   (tagged to the customer)
--                         Cr 1020 Bank                  excess refunded (bank only, if any)
--
--   Invoice balance = invoice total
--                     - refunds that reduced the receivable (damaged returns, approved)
--                     - payments applied to it (posted, not voided)
--
--   Damaged refund on a CREDIT sale (approve_sales_return):
--     the refund first reduces what the customer still owes on that invoice.
--     Anything beyond that was already paid by the customer, so it is paid back
--     in cash or by bank transfer (the owner chooses when approving).
--       Dr 4100 Sales Returns   refund
--         Cr 1100 Accounts Receivable   part that reduces the balance
--         Cr 1010 Cash | 1020 Bank      part paid back to the customer
-- =====================================================================

-- ---------- 0. Guards ---------------------------------------------------
do $$
declare
  c record;
begin
  for c in
    select conname, pg_get_constraintdef(oid) as def
    from pg_constraint
    where conrelid = 'public.journal_entries'::regclass and contype = 'c'
      and pg_get_constraintdef(oid) ilike '%source%'
  loop
    if c.def not ilike '%customer_payment%' then
      raise exception 'journal_entries has a rule on its source column (%: %). Send me this text so I can extend it to allow customer_payment, then re-run.', c.conname, c.def;
    end if;
  end loop;
end $$;

-- ---------- 1. Tables ---------------------------------------------------
create sequence if not exists public.customer_payment_no_seq;

create table if not exists public.customer_payments (
  id               uuid primary key default gen_random_uuid(),
  payment_no       bigint not null unique default nextval('public.customer_payment_no_seq'),
  payment_date     date not null,
  customer_id      uuid not null references public.customers(id),
  method           text not null check (method in ('cash','bank')),
  amount           numeric(14,2) not null check (amount > 0),          -- money received
  applied          numeric(14,2) not null check (applied > 0),         -- settles invoices
  excess_refunded  numeric(14,2) not null default 0 check (excess_refunded >= 0),
  reference        text,                                               -- bank reference / receipt no.
  notes            text,
  status           text not null default 'Posted' check (status in ('Posted','Voided')),
  journal_entry_id uuid references public.journal_entries(id),
  void_entry_id    uuid references public.journal_entries(id),
  void_reason      text,
  created_by       uuid references auth.users(id) default auth.uid(),
  created_at       timestamptz not null default now(),
  check (amount = applied + excess_refunded),
  check (excess_refunded = 0 or method = 'bank')
);
create index if not exists customer_payments_customer_idx on public.customer_payments (customer_id);

create table if not exists public.customer_payment_allocations (
  id         uuid primary key default gen_random_uuid(),
  payment_id uuid not null references public.customer_payments(id),
  sale_id    uuid not null references public.sales(id),
  amount     numeric(14,2) not null check (amount > 0),
  unique (payment_id, sale_id)
);
create index if not exists customer_payment_allocations_sale_idx on public.customer_payment_allocations (sale_id);

-- How a damaged refund was settled
alter table public.sales_returns add column if not exists refund_ar     numeric(14,2) not null default 0;
alter table public.sales_returns add column if not exists refund_paid   numeric(14,2) not null default 0;
alter table public.sales_returns add column if not exists refund_method text check (refund_method in ('cash','bank'));

-- Refunds approved before this file existed: credit sales reduced the receivable,
-- cash / bank sales were paid back the way the sale was paid.
update public.sales_returns r
   set refund_ar = r.refund_amount
  from public.sales s
 where s.id = r.sale_id and s.payment_type = 'credit'
   and r.resolution = 'refund' and r.status = 'Approved'
   and r.refund_ar = 0 and r.refund_paid = 0;

update public.sales_returns r
   set refund_paid = r.refund_amount,
       refund_method = case s.payment_type when 'cash' then 'cash' else 'bank' end
  from public.sales s
 where s.id = r.sale_id and s.payment_type in ('cash','bank')
   and r.resolution = 'refund' and r.status = 'Approved'
   and r.refund_paid = 0;

-- ---------- 2. Balances and ledger --------------------------------------
-- One row per posted credit invoice, with what is still owed on it.
create or replace view public.credit_invoice_balances
with (security_invoker = true) as
select y.*,
       case when y.balance <= 0 then 'Paid' when y.paid > 0 then 'Partial' else 'Unpaid' end as pay_status
from (
  select x.*, x.total - x.returned - x.paid as balance
  from (
    select s.id as sale_id, s.customer_id, s.sale_no, s.invoice_no, s.invoice_date, s.due_date, s.total,
           coalesce((select sum(r.refund_ar) from public.sales_returns r
                      where r.sale_id = s.id and r.status = 'Approved'), 0) as returned,
           coalesce((select sum(a.amount) from public.customer_payment_allocations a
                       join public.customer_payments p on p.id = a.payment_id
                      where a.sale_id = s.id and p.status = 'Posted'), 0) as paid
    from public.sales s
    where s.payment_type = 'credit' and s.status = 'Posted'
  ) x
) y;

-- Customer ledger: every invoice (debit), refund-return and payment (credit),
-- oldest first, with a running balance that matches customer_ar.
create or replace view public.customer_ledger
with (security_invoker = true) as
select t.*,
       sum(t.debit - t.credit) over (
         partition by t.customer_id
         order by t.entry_date, t.ord, t.created_at, t.doc_no
         rows between unbounded preceding and current row
       ) as balance
from (
  select s.customer_id, s.invoice_date as entry_date, 1 as ord,
         'INV-' || lpad(s.invoice_no::text, 4, '0') as doc_no,
         'invoice'::text as kind,
         ('Invoice ' || 'INV-' || lpad(s.invoice_no::text, 4, '0') ||
            case when s.due_date is not null then ' (due ' || to_char(s.due_date, 'YYYY-MM-DD') || ')' else '' end) as description,
         s.total as debit, 0::numeric as credit, s.id as sale_id, s.created_at
  from public.sales s
  where s.payment_type = 'credit' and s.status = 'Posted'

  union all

  select s.customer_id, r.return_date, 2,
         'RT-' || lpad(r.return_no::text, 4, '0'),
         'return',
         ('Damaged return RT-' || lpad(r.return_no::text, 4, '0') || ' on INV-' || lpad(s.invoice_no::text, 4, '0') ||
            case when r.refund_paid > 0
                 then ' (PHP ' || to_char(r.refund_paid, 'FM999,999,990.00') || ' of the refund was already paid, returned by ' ||
                      case r.refund_method when 'bank' then 'bank transfer' else 'cash' end || ')'
                 else '' end),
         0::numeric, r.refund_ar, s.id, r.created_at
  from public.sales_returns r
  join public.sales s on s.id = r.sale_id
  where s.payment_type = 'credit' and r.status = 'Approved' and r.resolution = 'refund'
    and (r.refund_ar > 0 or r.refund_paid > 0)

  union all

  select p.customer_id, p.payment_date, 3,
         'PR-' || lpad(p.payment_no::text, 4, '0'),
         'payment',
         ('Payment received by ' || case p.method when 'bank' then 'bank transfer' else 'cash' end ||
            coalesce(' (ref ' || p.reference || ')', '') ||
            case when p.excess_refunded > 0
                 then ', excess of PHP ' || to_char(p.excess_refunded, 'FM999,999,990.00') || ' returned by bank transfer'
                 else '' end),
         0::numeric, p.applied, null::uuid, p.created_at
  from public.customer_payments p
  where p.status = 'Posted'
) t;

-- ---------- 3. Internal: what is still owed on one invoice ---------------
create or replace function public.sale_open_balance(p_sale uuid)
returns numeric
language sql stable security definer set search_path = public as $$
  select s.total
         - coalesce((select sum(r.refund_ar) from public.sales_returns r
                      where r.sale_id = s.id and r.status = 'Approved'), 0)
         - coalesce((select sum(a.amount) from public.customer_payment_allocations a
                       join public.customer_payments p on p.id = a.payment_id
                      where a.sale_id = s.id and p.status = 'Posted'), 0)
  from public.sales s
  where s.id = p_sale;
$$;

-- ---------- 4. Record a customer payment (owner only) --------------------
-- p_allocations: optional [{sale_id, amount}]. Leave it out to settle the oldest due date first.
-- p_refund_excess: bank transfers only. Confirms that any amount above what is owed
--   is sent back by bank transfer (recorded in the same entry).
create or replace function public.record_customer_payment(
  p_date date, p_customer uuid, p_method text, p_amount numeric, p_reference text, p_notes text,
  p_allocations jsonb default null, p_refund_excess boolean default false
) returns uuid
language plpgsql security definer set search_path = public as $$
declare
  v_id uuid := gen_random_uuid();
  v_no bigint;
  v_today date := (now() at time zone 'Asia/Manila')::date;
  v_date date := coalesce(p_date, (now() at time zone 'Asia/Manila')::date);
  v_amount numeric(14,2);
  v_ref text := nullif(trim(coalesce(p_reference, '')), '');
  v_label text;
  v_cust text;
  v_open numeric(14,2);
  v_left numeric(14,2);
  v_take numeric(14,2);
  v_applied numeric(14,2) := 0;
  v_excess numeric(14,2);
  v_alloc jsonb := '[]'::jsonb;
  v_a jsonb;
  v_sale uuid;
  v_amt numeric(14,2);
  v_bal numeric(14,2);
  v_seen uuid[] := '{}';
  inv record;
  v_debit text;
  v_lines jsonb;
  v_je uuid;
begin
  if not public.is_owner() then raise exception 'Only the owner can record a customer payment.'; end if;
  if p_method is null or p_method not in ('cash','bank') then raise exception 'Choose cash or bank transfer.'; end if;
  v_amount := round(coalesce(p_amount, 0), 2);
  if v_amount <= 0 then raise exception 'Enter the amount received.'; end if;
  if v_date > v_today then raise exception 'The payment date cannot be in the future.'; end if;
  if p_method = 'bank' and v_ref is null then
    raise exception 'Enter the bank transfer reference number so the payment can be traced.';
  end if;

  select name into v_cust from public.customers where id = p_customer;
  if not found then raise exception 'Customer not found.'; end if;

  -- lock this customer's credit invoices so two payments cannot settle the same balance
  perform 1 from public.sales
   where customer_id = p_customer and payment_type = 'credit' and status = 'Posted' for update;

  select coalesce(sum(balance), 0) into v_open
  from public.credit_invoice_balances where customer_id = p_customer and balance > 0;
  if v_open <= 0 then raise exception 'This customer has nothing outstanding.'; end if;

  if p_allocations is null or jsonb_typeof(p_allocations) <> 'array' or jsonb_array_length(p_allocations) = 0 then
    -- oldest due date first
    v_left := v_amount;
    for inv in
      select sale_id, balance from public.credit_invoice_balances
      where customer_id = p_customer and balance > 0
      order by due_date nulls last, invoice_no
    loop
      exit when v_left <= 0;
      v_take := least(v_left, inv.balance);
      v_alloc := v_alloc || jsonb_build_array(jsonb_build_object('sale_id', inv.sale_id, 'amount', v_take));
      v_left := v_left - v_take;
    end loop;
  else
    for v_a in select * from jsonb_array_elements(p_allocations) loop
      v_sale := nullif(v_a->>'sale_id', '')::uuid;
      v_amt := round(coalesce(nullif(v_a->>'amount', '')::numeric, 0), 2);
      if v_amt <= 0 then continue; end if;
      if v_sale = any(v_seen) then raise exception 'The same invoice appears twice.'; end if;
      v_seen := v_seen || v_sale;
      select balance into v_bal from public.credit_invoice_balances
       where sale_id = v_sale and customer_id = p_customer;
      if not found or v_bal <= 0 then raise exception 'One of the invoices is not open for this customer.'; end if;
      if v_amt > v_bal then
        raise exception 'INV-%: only % is still owed, so % cannot be applied.',
          lpad((select invoice_no from public.sales where id = v_sale)::text, 4, '0'), v_bal, v_amt;
      end if;
      v_alloc := v_alloc || jsonb_build_array(jsonb_build_object('sale_id', v_sale, 'amount', v_amt));
    end loop;
  end if;

  select coalesce(sum((e->>'amount')::numeric), 0) into v_applied from jsonb_array_elements(v_alloc) e;
  if v_applied <= 0 then raise exception 'Apply the payment to at least one invoice.'; end if;
  if v_applied > v_amount then raise exception 'More was applied to invoices than was received.'; end if;

  v_excess := v_amount - v_applied;
  if v_excess > 0 then
    if p_method <> 'bank' then
      raise exception 'The customer owes % but % was received. Cash overpayments are not accepted and no customer credit is kept. Take only what is owed.', v_open, v_amount;
    end if;
    if v_applied < v_open then
      raise exception 'Apply the payment to every open invoice before treating the rest as an overpayment.';
    end if;
    if not coalesce(p_refund_excess, false) then
      raise exception 'The customer owes % but % was received. Overpayments are not kept as customer credit: confirm that the excess of % is refunded by bank transfer.', v_open, v_amount, v_excess;
    end if;
  end if;

  v_no := nextval('public.customer_payment_no_seq');
  v_label := 'PR-' || lpad(v_no::text, 4, '0');

  insert into public.customer_payments
    (id, payment_no, payment_date, customer_id, method, amount, applied, excess_refunded, reference, notes)
  values
    (v_id, v_no, v_date, p_customer, p_method, v_amount, v_applied, v_excess, v_ref,
     nullif(trim(coalesce(p_notes, '')), ''));

  insert into public.customer_payment_allocations (payment_id, sale_id, amount)
  select v_id, (e->>'sale_id')::uuid, (e->>'amount')::numeric from jsonb_array_elements(v_alloc) e;

  v_debit := case p_method when 'bank' then '1020' else '1010' end;
  v_lines := jsonb_build_array(
    jsonb_build_object('account_code', v_debit, 'debit', v_amount, 'memo', v_label || ' received'),
    jsonb_build_object('account_code', '1100', 'credit', v_applied, 'memo', v_label || ' applied to invoices',
                       'customer_id', p_customer)
  );
  if v_excess > 0 then
    v_lines := v_lines || jsonb_build_array(
      jsonb_build_object('account_code', '1020', 'credit', v_excess,
                         'memo', v_label || ' overpayment refunded by bank transfer'));
  end if;

  v_je := public.post_module_entry(v_date, v_label,
            'Payment ' || v_label || ' - ' || v_cust,
            v_lines, 'customer_payment', v_id);

  update public.customer_payments set journal_entry_id = v_je where id = v_id;
  return v_id;
end;
$$;

-- ---------- 5. Void a customer payment (owner only) ----------------------
create or replace function public.void_customer_payment(p_id uuid, p_reason text)
returns void
language plpgsql security definer set search_path = public as $$
declare
  p public.customer_payments;
  v_rev uuid;
begin
  if not public.is_owner() then raise exception 'Only the owner can void a payment.'; end if;
  if coalesce(trim(p_reason), '') = '' then raise exception 'Please give a reason.'; end if;
  select * into p from public.customer_payments where id = p_id for update;
  if not found then raise exception 'Payment not found.'; end if;
  if p.status <> 'Posted' then raise exception 'This payment was already voided.'; end if;

  if exists (
    select 1
    from public.customer_payment_allocations a
    join public.sales_returns r on r.sale_id = a.sale_id
    where a.payment_id = p_id and r.status = 'Approved' and r.refund_paid > 0
  ) then
    raise exception 'A refund was already paid back on an invoice this payment settled, so it cannot be voided.';
  end if;

  v_rev := public.reverse_module_entry(p.journal_entry_id, p_reason, 'customer_payment', p_id);
  update public.customer_payments
     set status = 'Voided', void_entry_id = v_rev, void_reason = trim(p_reason)
   where id = p_id;
end;
$$;

-- ---------- 6. Void a sale: now also refuses sales that have payments ----
create or replace function public.void_sale(p_id uuid, p_reason text)
returns void
language plpgsql security definer set search_path = public as $$
declare
  s public.sales;
  v_rev uuid;
begin
  if not public.is_owner() then raise exception 'Only the owner can void a sale.'; end if;
  if coalesce(trim(p_reason), '') = '' then raise exception 'Please give a reason.'; end if;
  select * into s from public.sales where id = p_id for update;
  if not found then raise exception 'Sale not found.'; end if;
  if s.status <> 'Posted' then raise exception 'Only an invoiced sale can be voided.'; end if;

  if exists (select 1 from public.sales_returns where sale_id = p_id and status <> 'Rejected') then
    raise exception 'A return or exchange was recorded on this sale, so it cannot be voided.';
  end if;

  if exists (
    select 1 from public.customer_payment_allocations a
    join public.customer_payments p on p.id = a.payment_id
    where a.sale_id = p_id and p.status = 'Posted'
  ) then
    raise exception 'A customer payment was recorded against this invoice. Void that payment first.';
  end if;

  if public.restore_stock_for('sale', p_id, 'sale_void') = 0 and s.cost_total > 0 then
    raise exception 'The stock taken by this sale could not be found, so it cannot be voided safely.';
  end if;

  v_rev := public.reverse_module_entry(s.journal_entry_id, p_reason, 'sale', p_id);
  update public.sales set status = 'Voided', void_entry_id = v_rev, void_reason = trim(p_reason) where id = p_id;
end;
$$;

-- ---------- 7. Approve a return: refunds on credit sales use the balance --
-- The old one-argument version must go, or a call with one argument is ambiguous.
drop function if exists public.approve_sales_return(uuid);

create or replace function public.approve_sales_return(p_id uuid, p_refund_method text default null)
returns void
language plpgsql security definer set search_path = public as $$
declare
  r public.sales_returns;
  s public.sales;
  v_ret_cost numeric(14,2);
  v_out_cost numeric(14,2) := 0;
  v_label text;
  v_lines jsonb := '[]'::jsonb;
  v_je uuid;
  v_bal numeric(14,2);
  v_ar numeric(14,2) := 0;
  v_paid numeric(14,2) := 0;
  v_method text;
begin
  if not public.is_owner() then raise exception 'Only the owner can approve returns.'; end if;
  select * into r from public.sales_returns where id = p_id for update;
  if not found then raise exception 'Return not found.'; end if;
  if r.status <> 'Pending' then raise exception 'This return was already decided.'; end if;
  select * into s from public.sales where id = r.sale_id for update;

  v_label := 'RT-' || lpad(r.return_no::text, 4, '0');
  v_ret_cost := round(r.unit_cost * r.qty, 2);

  -- the replacement goes out first, so it can never be the piece that just came back
  if r.resolution in ('exchange','replace') then
    v_out_cost := public.stock_out('finished', r.repl_size_id, r.qty, r.return_date, 'sales_return', r.id);
  end if;

  if r.resolution = 'exchange' then
    perform public.stock_in('finished', null, null, r.size_id, r.return_date, r.qty, v_ret_cost, 'sales_return', r.id);
    if v_ret_cost > 0 then
      v_lines := v_lines || jsonb_build_array(
        jsonb_build_object('account_code', '1330', 'debit',  v_ret_cost, 'memo', v_label || ' piece returned'),
        jsonb_build_object('account_code', '5000', 'credit', v_ret_cost, 'memo', v_label || ' piece returned'));
    end if;
    if v_out_cost > 0 then
      v_lines := v_lines || jsonb_build_array(
        jsonb_build_object('account_code', '5000', 'debit',  v_out_cost, 'memo', v_label || ' replacement given'),
        jsonb_build_object('account_code', '1330', 'credit', v_out_cost, 'memo', v_label || ' replacement given'));
    end if;

  elsif r.resolution = 'replace' then
    if v_ret_cost > 0 then
      v_lines := v_lines || jsonb_build_array(
        jsonb_build_object('account_code', '5130', 'debit',  v_ret_cost, 'memo', v_label || ' damaged piece written off'),
        jsonb_build_object('account_code', '5000', 'credit', v_ret_cost, 'memo', v_label || ' damaged piece written off'));
    end if;
    if v_out_cost > 0 then
      v_lines := v_lines || jsonb_build_array(
        jsonb_build_object('account_code', '5000', 'debit',  v_out_cost, 'memo', v_label || ' replacement given'),
        jsonb_build_object('account_code', '1330', 'credit', v_out_cost, 'memo', v_label || ' replacement given'));
    end if;

  else -- refund
    if s.payment_type = 'credit' then
      -- first lower what the customer still owes on this invoice
      v_bal := greatest(coalesce(public.sale_open_balance(s.id), 0), 0);
      v_ar := least(r.refund_amount, v_bal);
      v_paid := r.refund_amount - v_ar;
      if v_paid > 0 then
        -- the rest was already paid by the customer: it goes back as cash or bank transfer
        if p_refund_method is null or p_refund_method not in ('cash','bank') then
          raise exception 'The customer already paid % of this refund. Choose whether it is paid back in cash or by bank transfer.', v_paid;
        end if;
        v_method := p_refund_method;
      end if;
    else
      v_ar := 0;
      v_paid := r.refund_amount;
      v_method := case s.payment_type when 'cash' then 'cash' else 'bank' end;
    end if;

    v_lines := v_lines || jsonb_build_array(
      jsonb_build_object('account_code', '4100', 'debit', r.refund_amount, 'memo', v_label || ' damaged return'));
    if v_ar > 0 then
      v_lines := v_lines || jsonb_build_array(
        jsonb_build_object('account_code', '1100', 'credit', v_ar, 'memo', v_label || ' reduces balance',
                           'customer_id', s.customer_id));
    end if;
    if v_paid > 0 then
      v_lines := v_lines || jsonb_build_array(
        jsonb_build_object('account_code', case v_method when 'cash' then '1010' else '1020' end,
                           'credit', v_paid, 'memo', v_label || ' refund paid back'));
    end if;
    if v_ret_cost > 0 then
      v_lines := v_lines || jsonb_build_array(
        jsonb_build_object('account_code', '5130', 'debit',  v_ret_cost, 'memo', v_label || ' damaged piece written off'),
        jsonb_build_object('account_code', '5000', 'credit', v_ret_cost, 'memo', v_label || ' damaged piece written off'));
    end if;
  end if;

  if jsonb_array_length(v_lines) > 0 then
    v_je := public.post_module_entry(r.return_date, v_label,
              'Return / exchange ' || v_label || ' on INV-' || lpad(s.invoice_no::text, 4, '0'),
              v_lines, 'sales_return', r.id);
  end if;

  update public.sales_returns
     set status = 'Approved', journal_entry_id = v_je, decided_by = auth.uid(), decided_at = now(),
         refund_ar = v_ar, refund_paid = v_paid, refund_method = v_method
   where id = p_id;
end;
$$;

-- ---------- 8. Security ------------------------------------------------
alter table public.customer_payments   enable row level security;
alter table public.customer_payment_allocations enable row level security;

drop policy if exists customer_payments_read on public.customer_payments;
create policy customer_payments_read on public.customer_payments for select using (public.is_active_user());
drop policy if exists customer_payment_allocations_read on public.customer_payment_allocations;
create policy customer_payment_allocations_read on public.customer_payment_allocations for select using (public.is_active_user());

revoke insert, update, delete on public.customer_payments, public.customer_payment_allocations from anon, authenticated;
