-- =====================================================================
-- Chinito Scento - credit sale due date counts from the SALE date.
-- Safe to re-run. Run in the Supabase SQL Editor AFTER file 09_sales.
--
-- Before: approving a credit sale used TODAY's date (Manila) for everything, so a sale
--         dated Jul 12 with 15-day terms, approved on Nov 11, fell due Nov 26.
-- Now:    approve_credit_sale uses the sale's own date (sales.sale_date), so the
--         invoice date, journal entry, stock movement and due date all follow it.
--         Jul 12 + 15 days = Jul 27.
--
-- Only approve_credit_sale changes. Same parameters, so the app call and permissions
-- are untouched. Credit sales that were already approved are NOT changed.
-- When the owner approves on the same day the sale was entered, nothing differs.
-- =====================================================================

create or replace function public.approve_credit_sale(p_id uuid, p_date date default null)
returns void
language plpgsql security definer set search_path = public as $$
declare
  s public.sales;
  cc public.customer_credit;
  v_out numeric := 0;
  v_date date;
begin
  if not public.is_owner() then raise exception 'Only the owner can approve credit sales.'; end if;
  select * into s from public.sales where id = p_id for update;
  if not found then raise exception 'Sale not found.'; end if;
  if s.status <> 'Pending' or s.payment_type <> 'credit' then
    raise exception 'This is not a credit sale waiting for approval.';
  end if;

  -- The sale is booked on ITS OWN date, not on the day the owner clicks Approve.
  -- This drives the invoice date, the journal entry, the stock movement and the due date
  -- (sale date + the customer's credit days).
  v_date := coalesce(p_date, s.sale_date, (now() at time zone 'Asia/Manila')::date);

  select * into cc from public.customer_credit where customer_id = s.customer_id;
  if not found or not cc.credit_allowed then
    raise exception 'This customer is no longer allowed credit.';
  end if;

  select coalesce(sum(l.debit - l.credit), 0) into v_out
  from public.journal_lines l join public.accounts a on a.id = l.account_id
  where a.code = '1100' and l.customer_id = s.customer_id;

  -- a limit of 0 means no limit has been set
  if cc.credit_limit > 0 and v_out + s.total > cc.credit_limit then
    raise exception 'Approving this would bring the customer to % against a credit limit of %.',
      round(v_out + s.total, 2), cc.credit_limit;
  end if;

  perform public.post_sale_internal(p_id, v_date);
  update public.sales set decided_by = auth.uid(), decided_at = now() where id = p_id;
end;
$$;

revoke all on function public.approve_credit_sale(uuid,date) from public, anon;
grant execute on function public.approve_credit_sale(uuid,date) to authenticated;

notify pgrst, 'reload schema';
