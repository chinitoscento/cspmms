-- 37_order_requests.sql
-- Public order requests for the Chinito Scento website.
-- Safe to run once. Creates NEW objects only; changes nothing in your existing tables.

-- 1. Tables ---------------------------------------------------------------
create table if not exists public.order_requests (
  id             uuid primary key default gen_random_uuid(),
  request_no     bigint generated always as identity,
  name           text not null,
  contact        text not null,
  contact_digits text not null,
  email          text,
  address        text,
  notes          text,
  ref_code       text,
  status         text not null default 'New'
                 check (status in ('New','Contacted','Confirmed','Fulfilled','Cancelled')),
  paid           boolean not null default false,
  paid_at        timestamptz,
  created_at     timestamptz not null default now(),
  updated_at     timestamptz not null default now()
);

create table if not exists public.order_request_lines (
  id            uuid primary key default gen_random_uuid(),
  request_id    uuid not null references public.order_requests(id) on delete cascade,
  size_id       uuid not null references public.perfume_sizes(id),
  perfume_name  text not null,
  size_ml       numeric not null,
  unit_price    numeric not null,
  qty           integer not null check (qty between 1 and 20)
);

create index if not exists order_requests_contact_idx on public.order_requests (contact_digits, created_at desc);
create index if not exists order_request_lines_req_idx on public.order_request_lines (request_id);

create or replace function public.cs_touch_order_request() returns trigger
language plpgsql as $$
begin
  new.updated_at := now();
  if new.paid and (tg_op = 'INSERT' or not old.paid) then new.paid_at := now(); end if;
  if not new.paid then new.paid_at := null; end if;
  return new;
end $$;

drop trigger if exists cs_touch_order_request on public.order_requests;
create trigger cs_touch_order_request before insert or update on public.order_requests
for each row execute function public.cs_touch_order_request();

-- 2. Security: public cannot read or write these tables directly ------------
alter table public.order_requests      enable row level security;
alter table public.order_request_lines enable row level security;

drop policy if exists staff_all on public.order_requests;
create policy staff_all on public.order_requests
  for all to authenticated using (true) with check (true);

drop policy if exists staff_all on public.order_request_lines;
create policy staff_all on public.order_request_lines
  for all to authenticated using (true) with check (true);

-- 3. Internal helper: which sizes may be sold online ------------------------
-- Active size, price > 0, perfume not discontinued/inactive/archived.
-- in_stock is true when any finished-goods layer for that size has quantity left.
create or replace function public.cs_sellable_sizes()
returns table (size_id uuid, perfume_name text, size_ml numeric, price numeric, in_stock boolean)
language sql stable security definer set search_path = public as $$
  select ps.id, p.name, ps.size_ml, ps.price,
         exists (select 1 from stock_layers sl
                 where sl.size_id = ps.id and sl.qty_remaining > 0)
  from perfume_sizes ps
  join perfumes p on p.id = ps.perfume_id
  where ps.is_active
    and ps.price is not null and ps.price > 0
    and lower(coalesce(p.status, '')) not in ('discontinued','inactive','archived');
$$;
revoke all on function public.cs_sellable_sizes() from public, anon, authenticated;

-- 4. Public function: the catalog ------------------------------------------
create or replace function public.public_catalog()
returns table (size_id uuid, perfume_name text, size_ml numeric, price numeric, in_stock boolean)
language sql stable security definer set search_path = public as $$
  select size_id, perfume_name, size_ml, price, in_stock
  from cs_sellable_sizes()
  order by perfume_name, size_ml;
$$;
revoke all on function public.public_catalog() from public;
grant execute on function public.public_catalog() to anon, authenticated;

-- 5. Public function: send an order request ----------------------------------
create or replace function public.submit_order_request(
  p_name    text,
  p_contact text,
  p_email   text,
  p_address text,
  p_notes   text,
  p_items   jsonb,
  p_hp      text default '',
  p_ref     text default null
) returns bigint
language plpgsql security definer set search_path = public as $$
declare
  v_id      uuid;
  v_no      bigint;
  v_digits  text;
  v_want    int;
  v_got     int;
begin
  -- spam trap: bots fill the hidden field. Pretend it worked.
  if coalesce(trim(p_hp), '') <> '' then return 0; end if;

  if coalesce(trim(p_name), '') = '' then raise exception 'Please enter your name.'; end if;
  v_digits := regexp_replace(coalesce(p_contact, ''), '\D', '', 'g');
  if length(v_digits) < 7 then raise exception 'Please enter a valid mobile number.'; end if;
  if length(p_name) > 120 or length(coalesce(p_contact,'')) > 40
     or length(coalesce(p_email,'')) > 160 or length(coalesce(p_address,'')) > 500
     or length(coalesce(p_notes,'')) > 1000 then
    raise exception 'One of the fields is too long.';
  end if;
  if p_items is null or jsonb_typeof(p_items) <> 'array' or jsonb_array_length(p_items) = 0 then
    raise exception 'Please add at least one perfume to your order.';
  end if;
  if jsonb_array_length(p_items) > 30 then raise exception 'Too many items in one request.'; end if;

  -- server-side limit per phone number
  if (select count(*) from order_requests
      where contact_digits = v_digits and created_at > now() - interval '1 hour') >= 5 then
    raise exception 'Too many requests from this number. Please try again later.';
  end if;

  insert into order_requests (name, contact, contact_digits, email, address, notes, ref_code)
  values (trim(p_name), trim(p_contact), v_digits,
          nullif(trim(coalesce(p_email,'')), ''), nullif(trim(coalesce(p_address,'')), ''),
          nullif(trim(coalesce(p_notes,'')), ''), nullif(trim(coalesce(p_ref,'')), ''))
  returning id, request_no into v_id, v_no;

  -- prices and names always come from the server, never from the browser
  begin
    insert into order_request_lines (request_id, size_id, perfume_name, size_ml, unit_price, qty)
    select v_id, s.size_id, s.perfume_name, s.size_ml, s.price, i.q
    from (select (e->>'size_id')::uuid as sid, sum((e->>'qty')::int) as q
          from jsonb_array_elements(p_items) e group by 1) i
    join cs_sellable_sizes() s on s.size_id = i.sid and s.in_stock
    where i.q between 1 and 20;

    select count(distinct (e->>'size_id')::uuid) into v_want from jsonb_array_elements(p_items) e;
  exception when others then
    raise exception 'Some items in your order are not valid. Please refresh the page and try again.';
  end;

  select count(*) into v_got from order_request_lines where request_id = v_id;
  if v_got <> v_want then
    raise exception 'Some items in your order are sold out or no longer available. Please refresh the page and try again.';
  end if;

  return v_no;
end $$;
revoke all on function public.submit_order_request(text,text,text,text,text,jsonb,text,text) from public;
grant execute on function public.submit_order_request(text,text,text,text,text,jsonb,text,text) to anon, authenticated;
