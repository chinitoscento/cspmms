-- =====================================================================
-- Chinito Scento - Creditors: the third party you OWE is not the supplier.
-- Run in the Supabase SQL Editor AFTER file 15. Safe to re-run.
-- TAKE A BACKUP FIRST (Supabase > Database > Backups), and ideally try it on a copy.
--
-- What this does
--   * Adds a Creditors list (the third party you owe money to).
--   * Each CREDIT purchase now has a creditor. The supplier stays on the purchase
--     as the seller the goods came from, and is now optional.
--   * Payments are made to a creditor and are applied to that creditor's purchases.
--   * assign_purchase_creditor() fixes the purchases you already entered (for
--     example all of July) in one go, and logs every change in creditor_corrections.
--
-- What this does NOT do (on purpose)
--   * It posts NO journal entries and reverses NOTHING. Accounts Payable (2000),
--     the income statement and the balance sheet are exactly the same before and after.
--     The journal is locked and only records the seller on the payable line; who you
--     owe lives on the purchase and payment records, and that is what is corrected.
--   * It does not change amounts, dates, invoice numbers, stock or installment schedules.
--
-- Changed functions: record_purchase (new last parameter p_creditor) and
-- record_supplier_payment (first party parameter is now p_creditor).
-- The Procurement screen must be updated at the same time (Procurement.jsx).
-- =====================================================================

-- ---------- 1. Creditors ----------------------------------------------
create table if not exists public.creditors (
  id             uuid primary key default gen_random_uuid(),
  name           text not null check (char_length(trim(name)) > 0),
  contact_number text,
  notes          text,
  status         text not null default 'Active' check (status in ('Active','Inactive')),
  created_by     uuid references auth.users(id) default auth.uid(),
  created_at     timestamptz not null default now()
);
create unique index if not exists creditors_name_uq on public.creditors (lower(trim(name)));

alter table public.creditors enable row level security;
drop policy if exists creditors_read   on public.creditors;
drop policy if exists creditors_insert on public.creditors;
drop policy if exists creditors_update on public.creditors;
create policy creditors_read   on public.creditors for select using (public.is_active_user());
create policy creditors_insert on public.creditors for insert with check (public.is_owner());
create policy creditors_update on public.creditors for update using (public.is_owner()) with check (public.is_owner());
revoke delete on public.creditors from anon, authenticated;

-- ---------- 2. Creditor on purchases and payments ---------------------
alter table public.purchases
  add column if not exists creditor_id uuid references public.creditors(id);
create index if not exists purchases_creditor_idx on public.purchases (creditor_id);

alter table public.supplier_payments
  add column if not exists creditor_id uuid references public.creditors(id);
alter table public.supplier_payments alter column supplier_id drop not null;   -- new payments go to a creditor
create index if not exists supplier_payments_creditor_idx on public.supplier_payments (creditor_id);

-- ---------- 3. Correction log (append only) ---------------------------
create table if not exists public.creditor_corrections (
  id                uuid primary key default gen_random_uuid(),
  batch_id          uuid not null,
  corrected_at      timestamptz not null default now(),
  corrected_by      uuid references auth.users(id) default auth.uid(),
  purchase_id       uuid not null references public.purchases(id),
  seller_supplier_id uuid references public.suppliers(id),   -- the supplier on the purchase (unchanged)
  old_creditor_id   uuid references public.creditors(id),    -- null = none was recorded (the supplier was treated as the creditor)
  new_creditor_id   uuid not null references public.creditors(id),
  reason            text not null
);
create index if not exists creditor_corrections_purchase_idx on public.creditor_corrections (purchase_id);

create or replace function public.block_creditor_correction_changes()
returns trigger language plpgsql as $$
begin
  raise exception 'The creditor correction log cannot be edited or deleted.';
end;
$$;
drop trigger if exists cc_immutable on public.creditor_corrections;
create trigger cc_immutable before update or delete on public.creditor_corrections
  for each row execute function public.block_creditor_correction_changes();

alter table public.creditor_corrections enable row level security;
drop policy if exists creditor_corrections_read on public.creditor_corrections;
create policy creditor_corrections_read on public.creditor_corrections for select using (public.is_owner());
revoke insert, update, delete on public.creditor_corrections from anon, authenticated;

-- ---------- 4. Fix existing purchases (owner only) --------------------
-- Sets the creditor on posted CREDIT purchases. Payments already made against those
-- purchases move to the same creditor. All or nothing: if anything is wrong, nothing changes.
create or replace function public.assign_purchase_creditor(p_purchase_ids uuid[], p_creditor uuid, p_reason text)
returns int
language plpgsql security definer set search_path = public as $$
declare
  v_batch uuid := gen_random_uuid();
  v_n int := 0;
  v_want int;
  r record;
begin
  if not public.is_owner() then raise exception 'Only the owner can assign creditors.'; end if;
  if coalesce(trim(p_reason), '') = '' then raise exception 'Please give a reason.'; end if;
  if p_purchase_ids is null or cardinality(p_purchase_ids) = 0 then raise exception 'Choose at least one purchase.'; end if;
  perform 1 from public.creditors where id = p_creditor and status = 'Active';
  if not found then raise exception 'Choose an active creditor.'; end if;

  select count(distinct x) into v_want from unnest(p_purchase_ids) x;

  for r in
    select * from public.purchases where id = any (p_purchase_ids) order by purchase_no for update
  loop
    if r.payment_type <> 'credit' or r.status <> 'Posted' then
      raise exception 'PUR-% is not a posted credit purchase, so it cannot have a creditor.', lpad(r.purchase_no::text, 4, '0');
    end if;
    if r.creditor_id is distinct from p_creditor then
      insert into public.creditor_corrections (batch_id, purchase_id, seller_supplier_id, old_creditor_id, new_creditor_id, reason)
      values (v_batch, r.id, r.supplier_id, r.creditor_id, p_creditor, trim(p_reason));
      update public.purchases set creditor_id = p_creditor where id = r.id;
    end if;
    v_n := v_n + 1;
  end loop;

  if v_n <> v_want then raise exception 'Some of the chosen purchases were not found.'; end if;

  -- payments already made against these purchases belong to the same creditor
  update public.supplier_payments sp set creditor_id = p_creditor
   where sp.id in (select a.payment_id from public.payment_allocations a where a.purchase_id = any (p_purchase_ids));

  -- a payment that also covered a purchase outside this selection would be split between creditors
  if exists (
    select 1
    from public.payment_allocations a
    join public.supplier_payments sp on sp.id = a.payment_id
    join public.purchases p on p.id = a.purchase_id
    where sp.creditor_id = p_creditor and p.creditor_id is distinct from p_creditor
  ) then
    raise exception 'A payment you already made also covers purchases that are not in this selection. Include all the purchases that payment covered, then try again.';
  end if;

  return v_n;
end;
$$;

-- ---------- 5. Record a purchase (now with a creditor) ----------------
drop function if exists public.record_purchase(date,uuid,text,text,date,text,jsonb,int,date,numeric,numeric);

create or replace function public.record_purchase(
  p_date date, p_supplier uuid, p_reference text, p_payment text,
  p_due date, p_notes text, p_lines jsonb,
  p_installments int default null, p_first_due date default null,
  p_interest numeric default 0, p_interest_rate numeric default null,
  p_creditor uuid default null
) returns uuid
language plpgsql security definer set search_path = public as $$
declare
  v_id uuid := gen_random_uuid();
  v_no bigint;
  v_date date := coalesce(p_date, current_date);
  v_line jsonb;
  v_n int := 0;
  v_type text;
  v_mat public.materials;
  v_qty numeric;
  v_cost numeric;
  v_amt numeric(14,2);
  v_code text;
  v_desc text;
  v_total numeric(14,2) := 0;
  v_terms int := 0;
  v_sup text;
  v_cred text;
  v_cred_id uuid;
  v_due date;
  v_dr jsonb;
  v_cr_code text;
  v_label text;
  v_je uuid;
  v_ninst int := coalesce(p_installments, 0);
  v_first date;
  v_int numeric(14,2) := round(coalesce(p_interest, 0), 2);
  v_plan uuid;
  v_pp numeric;
  v_ii numeric;
  k int;
begin
  if not public.is_active_user() then raise exception 'Not allowed.'; end if;
  if p_payment not in ('cash','bank','credit') then raise exception 'Choose how this purchase is paid.'; end if;
  if p_lines is null or jsonb_typeof(p_lines) <> 'array' or jsonb_array_length(p_lines) < 1 then
    raise exception 'Add at least one line.';
  end if;

  -- the supplier is the seller; it is optional
  if p_supplier is not null then
    select name, payment_terms_days into v_sup, v_terms from public.suppliers where id = p_supplier;
    if not found then raise exception 'Supplier not found.'; end if;
  end if;

  -- the creditor is who you owe; a purchase on credit needs one
  if p_payment = 'credit' then
    if p_creditor is null then raise exception 'A purchase on credit needs a creditor (the party you owe).'; end if;
    select name into v_cred from public.creditors where id = p_creditor and status = 'Active';
    if not found then raise exception 'Creditor not found or inactive.'; end if;
    v_cred_id := p_creditor;
  end if;

  v_due := case when p_payment = 'credit' then coalesce(p_due, v_date + coalesce(v_terms, 0)) end;

  if v_ninst <> 0 then
    if p_payment <> 'credit' then raise exception 'Installments are only for purchases on credit.'; end if;
    if v_ninst < 2 or v_ninst > 60 then raise exception 'Choose between 2 and 60 installments.'; end if;
    v_first := coalesce(p_first_due, (v_date + interval '1 month')::date);
    if v_first < v_date then raise exception 'The first due date cannot be before the purchase date.'; end if;
    v_due := v_first;
  end if;
  if v_int < 0 then raise exception 'Interest cannot be negative.'; end if;
  if v_int > 0 and v_ninst = 0 then raise exception 'Interest can only be set on an installment plan.'; end if;

  v_no := nextval('public.purchase_no_seq');
  v_label := 'PUR-' || lpad(v_no::text, 4, '0');

  insert into public.purchases (id, purchase_no, purchase_date, supplier_id, creditor_id, reference, payment_type, due_date, notes, total)
  values (v_id, v_no, v_date, p_supplier, v_cred_id, nullif(trim(p_reference), ''), p_payment, v_due, nullif(trim(p_notes), ''), 0);

  for v_line in select * from jsonb_array_elements(p_lines) loop
    v_n := v_n + 1;
    v_type := v_line->>'line_type';
    v_qty := coalesce((v_line->>'qty')::numeric, 0);
    v_cost := coalesce((v_line->>'unit_cost')::numeric, 0);
    if v_qty <= 0 then raise exception 'Line %: quantity must be more than zero.', v_n; end if;
    if v_cost < 0 then raise exception 'Line %: cost cannot be negative.', v_n; end if;
    v_amt := round(v_qty * v_cost, 2);
    v_mat := null;

    if v_type = 'material' then
      select * into v_mat from public.materials
       where id = nullif(v_line->>'material_id', '')::uuid and is_active;
      if not found then raise exception 'Line %: choose a material.', v_n; end if;
      v_code := case v_mat.category when 'oil' then '1200' when 'solvent' then '1210' else '1220' end;
      v_desc := v_mat.name;
    elsif v_type = 'asset' then
      v_code := v_line->>'account_code';
      if v_code is null or v_code not in ('1400','1500') then
        raise exception 'Line %: an asset goes to Prepaid Expenses or Equipment.', v_n;
      end if;
      v_desc := nullif(trim(coalesce(v_line->>'description', '')), '');
      if v_desc is null then raise exception 'Line %: describe the item.', v_n; end if;
    elsif v_type = 'expense' then
      v_code := v_line->>'account_code';
      perform 1 from public.accounts
       where code = v_code and type = 'expense' and is_active and not is_system and code not in ('6060','6080');
      if not found then raise exception 'Line %: choose an expense account.', v_n; end if;
      v_desc := nullif(trim(coalesce(v_line->>'description', '')), '');
      if v_desc is null then raise exception 'Line %: describe the expense.', v_n; end if;
    else
      raise exception 'Line %: unknown line type.', v_n;
    end if;

    insert into public.purchase_lines (purchase_id, line_no, line_type, material_id, description, account_code, qty, unit_cost, amount)
    values (v_id, v_n, v_type, v_mat.id, v_desc, v_code, v_qty, v_cost, v_amt);

    if v_type = 'material' then
      perform public.stock_in('material', v_mat.id, null, null, v_date, v_qty, v_amt, 'purchase', v_id);
    end if;
    v_total := v_total + v_amt;
  end loop;

  if v_total <= 0 then raise exception 'The purchase total cannot be zero.'; end if;

  select jsonb_agg(jsonb_build_object('account_code', x.code, 'debit', x.amt, 'memo', 'Purchase ' || v_label) order by x.code)
    into v_dr
  from (
    select account_code as code, sum(amount) as amt
    from public.purchase_lines where purchase_id = v_id
    group by account_code having sum(amount) > 0
  ) x;

  v_cr_code := case p_payment when 'cash' then '1010' when 'bank' then '1020' else '2000' end;

  v_je := public.post_module_entry(
    v_date, v_label,
    'Purchase ' || v_label || coalesce(' from ' || v_sup, '') || coalesce(' - owed to ' || v_cred, ''),
    v_dr || jsonb_build_array(jsonb_build_object(
      'account_code', v_cr_code, 'credit', v_total,
      'memo', 'Purchase ' || v_label, 'supplier_id', p_supplier)),
    'procurement', v_id);

  update public.purchases set total = v_total, journal_entry_id = v_je where id = v_id;

  if v_ninst >= 2 then
    insert into public.purchase_installment_plans (purchase_id, installments, first_due, interest_total, interest_rate_pct)
    values (v_id, v_ninst, v_first, v_int, p_interest_rate)
    returning id into v_plan;
    v_pp := floor(v_total * 100 / v_ninst) / 100;
    v_ii := floor(v_int * 100 / v_ninst) / 100;
    for k in 1..v_ninst loop
      insert into public.purchase_installments (plan_id, purchase_id, seq, due_date, principal_due, interest_due)
      values (v_plan, v_id, k, (v_first + make_interval(months => k - 1))::date,
              case when k = v_ninst then v_total - v_pp * (v_ninst - 1) else v_pp end,
              case when k = v_ninst then v_int - v_ii * (v_ninst - 1) else v_ii end);
    end loop;
  end if;
  return v_id;
end;
$$;

-- ---------- 6. Pay a creditor (installment-aware) ---------------------
-- Same rules as before (interest first within each installment). The only differences:
--   * the party is a creditor, and only that creditor's purchases can be paid
--   * the journal lines keep the seller tag of each purchase paid, so the supplier tags
--     already on Accounts Payable keep balancing
drop function if exists public.record_supplier_payment(date,uuid,text,text,text,jsonb);

create or replace function public.record_supplier_payment(
  p_date date, p_creditor uuid, p_method text, p_reference text, p_notes text, p_allocations jsonb
) returns uuid
language plpgsql security definer set search_path = public as $$
declare
  v_id uuid := gen_random_uuid();
  v_no bigint;
  v_date date := coalesce(p_date, current_date);
  v_a jsonb;
  v_amt numeric(14,2);
  v_total numeric(14,2) := 0;
  v_prin numeric(14,2) := 0;
  v_int numeric(14,2) := 0;
  v_lp numeric(14,2);
  v_li numeric(14,2);
  v_key text;
  v_prin_by jsonb := '{}'::jsonb;
  v_int_by jsonb := '{}'::jsonb;
  v_pur public.purchases;
  v_plan public.purchase_installment_plans;
  v_has_plan boolean;
  v_paid numeric;
  v_limit numeric;
  v_left numeric;
  v_pay_i numeric;
  v_pay_p numeric;
  v_due_i numeric;
  v_due_p numeric;
  r record;
  g record;
  v_plans uuid[] := '{}';
  v_pl uuid;
  v_int_paid numeric;
  v_ip_code text;
  v_cred text;
  v_label text;
  v_lines jsonb;
  v_je uuid;
begin
  if not public.is_owner() then raise exception 'Only the owner can pay creditors.'; end if;
  if p_method not in ('cash','bank') then raise exception 'Choose cash or bank.'; end if;
  if p_allocations is null or jsonb_typeof(p_allocations) <> 'array' or jsonb_array_length(p_allocations) < 1 then
    raise exception 'Choose at least one purchase to pay.';
  end if;
  select name into v_cred from public.creditors where id = p_creditor;
  if not found then raise exception 'Creditor not found.'; end if;
  select interest_payable_code into v_ip_code from public.business_settings where id;

  v_no := nextval('public.payment_no_seq');
  v_label := 'PAY-' || lpad(v_no::text, 4, '0');

  insert into public.supplier_payments (id, pay_no, pay_date, supplier_id, creditor_id, amount, method, reference, notes)
  values (v_id, v_no, v_date, null, p_creditor, 0, p_method, nullif(trim(p_reference), ''), nullif(trim(p_notes), ''));

  for v_a in select * from jsonb_array_elements(p_allocations) loop
    v_amt := round(coalesce((v_a->>'amount')::numeric, 0), 2);
    if v_amt <= 0 then continue; end if;

    select * into v_pur from public.purchases where id = (v_a->>'purchase_id')::uuid for update;
    if not found or v_pur.payment_type <> 'credit' or v_pur.status <> 'Posted' then
      raise exception 'One of the purchases cannot be paid here.';
    end if;
    if v_pur.creditor_id is null then
      raise exception 'PUR-% has no creditor yet. Assign its creditor first.', lpad(v_pur.purchase_no::text, 4, '0');
    end if;
    if v_pur.creditor_id <> p_creditor then
      raise exception 'PUR-% is owed to a different creditor.', lpad(v_pur.purchase_no::text, 4, '0');
    end if;

    select * into v_plan from public.purchase_installment_plans where purchase_id = v_pur.id;
    v_has_plan := found;
    v_limit := v_pur.total + case when v_has_plan then v_plan.interest_total else 0 end;

    insert into public.payment_allocations (payment_id, purchase_id, amount) values (v_id, v_pur.id, v_amt);

    select coalesce(sum(a.amount), 0) into v_paid
    from public.payment_allocations a
    join public.supplier_payments sp on sp.id = a.payment_id and sp.status = 'Posted'
    where a.purchase_id = v_pur.id;
    if v_paid > v_limit then
      raise exception 'Payment for PUR-% is more than its balance.', lpad(v_pur.purchase_no::text, 4, '0');
    end if;

    v_lp := 0; v_li := 0;
    if v_has_plan then
      v_left := v_amt;
      for r in
        select i.id, i.interest_due, i.principal_due,
               coalesce(x.paid_i, 0) as paid_i, coalesce(x.paid_p, 0) as paid_p
        from public.purchase_installments i
        left join lateral (
          select sum(ip.interest_amt) as paid_i, sum(ip.principal_amt) as paid_p
          from public.installment_payments ip
          join public.supplier_payments sp on sp.id = ip.payment_id and sp.status = 'Posted'
          where ip.installment_id = i.id
        ) x on true
        where i.plan_id = v_plan.id
        order by i.seq
      loop
        exit when v_left <= 0;
        v_due_i := greatest(r.interest_due - r.paid_i, 0);
        v_due_p := greatest(r.principal_due - r.paid_p, 0);
        continue when v_due_i <= 0 and v_due_p <= 0;
        v_pay_i := least(v_left, v_due_i);
        v_left := v_left - v_pay_i;
        v_pay_p := least(v_left, v_due_p);
        v_left := v_left - v_pay_p;
        insert into public.installment_payments (payment_id, installment_id, principal_amt, interest_amt)
        values (v_id, r.id, v_pay_p, v_pay_i);
        v_lp := v_lp + v_pay_p;
        v_li := v_li + v_pay_i;
      end loop;
      if v_left > 0 then
        raise exception 'Payment for PUR-% is more than its balance.', lpad(v_pur.purchase_no::text, 4, '0');
      end if;
      if not (v_plan.id = any (v_plans)) then v_plans := v_plans || v_plan.id; end if;
    else
      v_lp := v_amt;
    end if;

    -- remember how much went to which seller, so the journal keeps its supplier tags
    v_key := coalesce(v_pur.supplier_id::text, '');
    v_prin_by := jsonb_set(v_prin_by, array[v_key], to_jsonb(coalesce((v_prin_by->>v_key)::numeric, 0) + v_lp), true);
    v_int_by  := jsonb_set(v_int_by,  array[v_key], to_jsonb(coalesce((v_int_by->>v_key)::numeric, 0) + v_li), true);
    v_prin := v_prin + v_lp;
    v_int := v_int + v_li;
    v_total := v_total + v_amt;
  end loop;

  if v_total <= 0 then raise exception 'Enter an amount to pay.'; end if;

  -- Interest being paid must already be on the books: bring the accrual up to date first.
  foreach v_pl in array v_plans loop
    perform public.accrue_plan_months(v_pl, (date_trunc('month', v_date::timestamp) - interval '1 day')::date);
    select coalesce(sum(ip.interest_amt), 0) into v_int_paid
    from public.installment_payments ip
    join public.supplier_payments sp on sp.id = ip.payment_id and sp.status = 'Posted'
    join public.purchase_installments i on i.id = ip.installment_id
    where i.plan_id = v_pl;
    perform public.accrue_plan_interest(v_pl, v_date, v_int_paid, v_date, 'catchup');
  end loop;

  v_lines := '[]'::jsonb;
  for g in select key as k, value::numeric as amt from jsonb_each_text(v_prin_by) loop
    if g.amt > 0 then
      v_lines := v_lines || jsonb_build_array(jsonb_build_object(
        'account_code', '2000', 'debit', g.amt, 'memo', 'Payment ' || v_label, 'supplier_id', nullif(g.k, '')));
    end if;
  end loop;
  for g in select key as k, value::numeric as amt from jsonb_each_text(v_int_by) loop
    if g.amt > 0 then
      v_lines := v_lines || jsonb_build_array(jsonb_build_object(
        'account_code', v_ip_code, 'debit', g.amt, 'memo', 'Interest paid ' || v_label, 'supplier_id', nullif(g.k, '')));
    end if;
  end loop;
  v_lines := v_lines || jsonb_build_array(jsonb_build_object(
    'account_code', case p_method when 'cash' then '1010' else '1020' end,
    'credit', v_total, 'memo', 'Payment ' || v_label));

  v_je := public.post_module_entry(
    v_date, v_label, 'Payment to ' || v_cred || ' (' || v_label || ')',
    v_lines, 'supplier_payment', v_id);

  update public.supplier_payments set amount = v_total, journal_entry_id = v_je where id = v_id;
  return v_id;
end;
$$;

-- ---------- 7. Open payables: now also shows the creditor --------------
-- The existing columns keep their names and order; creditor_id and creditor_name are added at the end.
-- supplier_name is still the seller. A credit purchase with no supplier now appears too.
create or replace view public.payables_open
with (security_invoker = true) as
select p.id as purchase_id, p.purchase_no, p.purchase_date,
       coalesce(nx.due_date, p.due_date) as due_date,
       p.supplier_id, s.name as supplier_name, p.reference,
       p.total + coalesce(pl.interest_total, 0) as total,
       coalesce(x.paid, 0) as paid,
       p.total + coalesce(pl.interest_total, 0) - coalesce(x.paid, 0) as balance,
       greatest(current_date - coalesce(nx.due_date, p.due_date), 0) as days_overdue,
       (pl.id is not null) as is_installment,
       pl.installments,
       coalesce(pl.interest_total, 0) as interest_total,
       coalesce(nx.balance, p.total - coalesce(x.paid, 0)) as next_due_balance,
       p.creditor_id, c.name as creditor_name
from public.purchases p
left join public.suppliers s on s.id = p.supplier_id
left join public.creditors c on c.id = p.creditor_id
left join public.purchase_installment_plans pl on pl.purchase_id = p.id
left join lateral (
  select sum(a.amount) as paid
  from public.payment_allocations a
  join public.supplier_payments sp on sp.id = a.payment_id and sp.status = 'Posted'
  where a.purchase_id = p.id
) x on true
left join lateral (
  select st.due_date, st.balance
  from public.installments_status st
  where st.purchase_id = p.id and st.balance > 0
  order by st.seq
  limit 1
) nx on true
where p.payment_type = 'credit' and p.status = 'Posted'
  and p.total + coalesce(pl.interest_total, 0) - coalesce(x.paid, 0) > 0;

-- ---------- 8. Permissions --------------------------------------------
revoke all on public.payables_open from public, anon;
grant select on public.payables_open to authenticated;

revoke all on function public.assign_purchase_creditor(uuid[],uuid,text) from public, anon;
revoke all on function public.record_purchase(date,uuid,text,text,date,text,jsonb,int,date,numeric,numeric,uuid) from public, anon;
revoke all on function public.record_supplier_payment(date,uuid,text,text,text,jsonb) from public, anon;
grant execute on function public.assign_purchase_creditor(uuid[],uuid,text) to authenticated;
grant execute on function public.record_purchase(date,uuid,text,text,date,text,jsonb,int,date,numeric,numeric,uuid) to authenticated;
grant execute on function public.record_supplier_payment(date,uuid,text,text,text,jsonb) to authenticated;
