-- =====================================================================
-- Chinito Scento - void for bottling and packaging runs.
-- Safe to re-run. Run in the Supabase SQL Editor.
--
-- How a void works (same idea as void_purchase):
--   * owner only, reason required
--   * everything the run took from stock goes back into the same FIFO layers
--   * the stock the run created is removed (only if none of it was used yet)
--   * the run's journal entry is reversed
--   * the run is kept and marked Voided (nothing is deleted)
--
-- It does not depend on the parameter list of record_bottling / record_packaging;
-- it uses the stock movement log, which both functions write with
-- ref_type = 'bottling' / 'packaging' and ref_id = the run id.
-- =====================================================================

-- ---------- 1. Void columns ------------------------------------------
alter table public.bottling_runs
  add column if not exists status text not null default 'Posted' check (status in ('Posted','Voided')),
  add column if not exists void_entry_id uuid references public.journal_entries(id),
  add column if not exists void_reason text,
  add column if not exists voided_at timestamptz,
  add column if not exists voided_by uuid references auth.users(id);

alter table public.packaging_runs
  add column if not exists status text not null default 'Posted' check (status in ('Posted','Voided')),
  add column if not exists void_entry_id uuid references public.journal_entries(id),
  add column if not exists void_reason text,
  add column if not exists voided_at timestamptz,
  add column if not exists voided_by uuid references auth.users(id);

-- ---------- 2. Internal helper: put back what a run took --------------
create or replace function public.restore_stock_for(p_ref_type text, p_ref_id uuid, p_void_type text)
returns int
language plpgsql security definer set search_path = public as $$
declare
  v_n int;
begin
  select count(*) into v_n
  from public.stock_movements
  where ref_type = p_ref_type and ref_id = p_ref_id and move_type = 'out';
  if v_n = 0 then
    return 0;
  end if;

  -- log first (the log is append-only), then give the quantity and cost back to the same layers
  insert into public.stock_movements (layer_id, move_date, move_type, qty, cost, ref_type, ref_id)
  select layer_id, current_date, 'void', -sum(qty), -sum(cost), p_void_type, p_ref_id
  from public.stock_movements
  where ref_type = p_ref_type and ref_id = p_ref_id and move_type = 'out'
  group by layer_id;

  update public.stock_layers l
     set qty_remaining  = l.qty_remaining  + s.q,
         cost_remaining = l.cost_remaining + s.c
    from (
      select layer_id, -sum(qty) as q, -sum(cost) as c
      from public.stock_movements
      where ref_type = p_ref_type and ref_id = p_ref_id and move_type = 'out'
      group by layer_id
    ) s
   where l.id = s.layer_id;

  return v_n;
end;
$$;

-- ---------- 3. Void bottling ------------------------------------------
create or replace function public.void_bottling(p_id uuid, p_reason text)
returns void
language plpgsql security definer set search_path = public as $$
declare
  r public.bottling_runs;
  v_rev uuid;
begin
  if not public.is_owner() then raise exception 'Only the owner can void a bottling run.'; end if;
  if coalesce(trim(p_reason), '') = '' then raise exception 'Please give a reason.'; end if;

  select * into r from public.bottling_runs where id = p_id for update;
  if not found then raise exception 'Bottling run not found.'; end if;
  if r.status <> 'Posted' then raise exception 'This bottling run is already voided.'; end if;

  perform 1 from public.stock_layers where source = 'bottling' and source_id = p_id for update;
  if not found then raise exception 'The stock record for this run was not found, so it cannot be voided safely.'; end if;

  if exists (
    select 1 from public.stock_layers
    where source = 'bottling' and source_id = p_id and qty_remaining <> qty_in
  ) then
    raise exception 'Some of these bottles were already packaged. Void the packaging first.';
  end if;

  if public.restore_stock_for('bottling', p_id, 'bottling_void') = 0 then
    raise exception 'The juice and materials used by this run could not be found, so it cannot be voided safely.';
  end if;

  -- remove the bottled stock this run created
  insert into public.stock_movements (layer_id, move_date, move_type, qty, cost, ref_type, ref_id)
  select id, current_date, 'void', -qty_remaining, -cost_remaining, 'bottling_void', p_id
  from public.stock_layers
  where source = 'bottling' and source_id = p_id and qty_remaining > 0;

  update public.stock_layers set qty_remaining = 0, cost_remaining = 0
  where source = 'bottling' and source_id = p_id;

  v_rev := public.reverse_module_entry(r.journal_entry_id, p_reason, 'bottling', p_id);

  update public.bottling_runs
     set status = 'Voided', void_entry_id = v_rev, void_reason = trim(p_reason),
         voided_at = now(), voided_by = auth.uid()
   where id = p_id;
end;
$$;

-- ---------- 4. Void packaging -----------------------------------------
create or replace function public.void_packaging(p_id uuid, p_reason text)
returns void
language plpgsql security definer set search_path = public as $$
declare
  r public.packaging_runs;
  v_rev uuid;
begin
  if not public.is_owner() then raise exception 'Only the owner can void a packaging run.'; end if;
  if coalesce(trim(p_reason), '') = '' then raise exception 'Please give a reason.'; end if;

  select * into r from public.packaging_runs where id = p_id for update;
  if not found then raise exception 'Packaging run not found.'; end if;
  if r.status <> 'Posted' then raise exception 'This packaging run is already voided.'; end if;

  perform 1 from public.stock_layers where source = 'packaging' and source_id = p_id for update;
  if not found then raise exception 'The stock record for this run was not found, so it cannot be voided safely.'; end if;

  if exists (
    select 1 from public.stock_layers
    where source = 'packaging' and source_id = p_id and qty_remaining <> qty_in
  ) then
    raise exception 'Some of these finished pieces were already used or sold, so this packaging run cannot be voided.';
  end if;

  if public.restore_stock_for('packaging', p_id, 'packaging_void') = 0 then
    raise exception 'The bottled perfume and materials used by this run could not be found, so it cannot be voided safely.';
  end if;

  -- remove the finished goods this run created
  insert into public.stock_movements (layer_id, move_date, move_type, qty, cost, ref_type, ref_id)
  select id, current_date, 'void', -qty_remaining, -cost_remaining, 'packaging_void', p_id
  from public.stock_layers
  where source = 'packaging' and source_id = p_id and qty_remaining > 0;

  update public.stock_layers set qty_remaining = 0, cost_remaining = 0
  where source = 'packaging' and source_id = p_id;

  v_rev := public.reverse_module_entry(r.journal_entry_id, p_reason, 'packaging', p_id);

  update public.packaging_runs
     set status = 'Voided', void_entry_id = v_rev, void_reason = trim(p_reason),
         voided_at = now(), voided_by = auth.uid()
   where id = p_id;
end;
$$;

-- ---------- 5. Permissions --------------------------------------------
revoke all on function public.restore_stock_for(text,uuid,text) from public, anon, authenticated;
revoke all on function public.void_bottling(uuid,text)  from public, anon;
revoke all on function public.void_packaging(uuid,text) from public, anon;
grant execute on function public.void_bottling(uuid,text)  to authenticated;
grant execute on function public.void_packaging(uuid,text) to authenticated;

notify pgrst, 'reload schema';
