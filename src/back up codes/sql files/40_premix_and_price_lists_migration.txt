-- PreMixes (250 ml bottled juice, no packaging step) and retail / reseller / distributor prices
-- Run once in the Supabase SQL editor. Safe to re-run.
--
-- PreMix
--   * A PreMix is an extra size on an existing perfume (so it uses that perfume's matured juice),
--     marked with perfume_sizes.is_premix.
--   * When a PreMix is bottled, the bottled units go straight into Finished Goods. The system books
--     a packaging run with no box or stickers (Dr 1330 Finished Goods, Cr 1320 Bottled), so voiding,
--     stock counts and sales work exactly as for any other finished item.
--   * To undo a PreMix bottling, void its packaging run first (Packaging page), then the bottling run.
--   * Stock lists show PreMix sizes under the group "PreMixes".
-- Prices
--   * perfume_sizes.price stays the retail (default) price. Each size can also have a reseller price and a
--     distributor price. A price of 0 on a list means "use the retail price".
--   * Each customer has a type: Retail (default), Reseller or Distributor. A sale uses the price list of its
--     customer's type (no customer = retail). The type at the time of the sale is saved on the sale
--     (sales.price_tier). Changing a price only affects new sales; existing invoices keep their prices.
--   * The list price on an invoice stays the retail price, so returns and exchanges work as before.

begin;

-- 1. Columns -------------------------------------------------------------------
alter table public.perfume_sizes
  add column if not exists is_premix boolean not null default false,
  add column if not exists reseller_price numeric not null default 0,
  add column if not exists distributor_price numeric not null default 0;

alter table public.perfume_sizes drop constraint if exists perfume_sizes_reseller_price_chk;
alter table public.perfume_sizes add constraint perfume_sizes_reseller_price_chk check (reseller_price >= 0);
alter table public.perfume_sizes drop constraint if exists perfume_sizes_distributor_price_chk;
alter table public.perfume_sizes add constraint perfume_sizes_distributor_price_chk check (distributor_price >= 0);

alter table public.customers
  add column if not exists category text not null default 'Retail';

alter table public.customers drop constraint if exists customers_category_chk;
alter table public.customers add constraint customers_category_chk check (category in ('Retail', 'Reseller', 'Distributor'));

alter table public.sales
  add column if not exists price_tier text not null default 'retail';

alter table public.sales drop constraint if exists sales_price_tier_chk;
alter table public.sales add constraint sales_price_tier_chk check (price_tier in ('retail', 'reseller', 'distributor'));

-- 2. Stock list: PreMix sizes sit in their own group ------------------------------
create or replace view public.inventory_status as
 WITH layer_sums AS (
         SELECT stock_layers.stock_kind,
            COALESCE(stock_layers.material_id, stock_layers.perfume_id, stock_layers.size_id) AS key_id,
            sum(stock_layers.qty_remaining) AS qty,
            sum(stock_layers.cost_remaining) AS val
           FROM stock_layers
          WHERE stock_layers.qty_remaining > 0::numeric
          GROUP BY stock_layers.stock_kind, (COALESCE(stock_layers.material_id, stock_layers.perfume_id, stock_layers.size_id))
        ), macerating AS (
         SELECT maceration_batches.perfume_id,
            sum(maceration_batches.expected_ml) AS ml
           FROM maceration_batches
          WHERE maceration_batches.status = 'Macerating'::text
          GROUP BY maceration_batches.perfume_id
        )
 SELECT 'material'::text AS kind,
    m.id AS item_id,
    m.code AS sku,
    m.name AS item_name,
    m.category AS group_name,
    NULL::numeric AS size_ml,
    m.unit,
    COALESCE(s.qty, 0::numeric) AS on_hand,
    COALESCE(s.val, 0::numeric) AS value,
    COALESCE(t.reorder_level, 0::numeric) AS reorder_level,
    NULL::numeric AS pipeline_units,
    NULL::numeric AS juice_ml,
    NULL::numeric AS macerating_ml,
        CASE
            WHEN COALESCE(t.reorder_level, 0::numeric) = 0::numeric THEN 'No level'::text
            WHEN COALESCE(s.qty, 0::numeric) <= 0::numeric THEN 'Out'::text
            WHEN COALESCE(s.qty, 0::numeric) <= t.reorder_level THEN 'Low'::text
            ELSE 'OK'::text
        END AS status,
    'Purchase'::text AS action
   FROM materials m
     LEFT JOIN layer_sums s ON s.stock_kind = 'material'::text AND s.key_id = m.id
     LEFT JOIN stock_thresholds t ON t.item_kind = 'material'::text AND t.item_id = m.id
  WHERE m.is_active
UNION ALL
 SELECT 'finished'::text AS kind,
    ps.id AS item_id,
    pf.sku,
    pf.name AS item_name,
    CASE WHEN ps.is_premix THEN 'PreMixes'::text ELSE COALESCE(pf.collection, 'Perfume'::text) END AS group_name,
    ps.size_ml::numeric AS size_ml,
    'pcs'::text AS unit,
    COALESCE(f.qty, 0::numeric) AS on_hand,
    COALESCE(f.val, 0::numeric) AS value,
    COALESCE(t.reorder_level, 0::numeric) AS reorder_level,
    COALESCE(b.qty, 0::numeric) AS pipeline_units,
    COALESCE(j.qty, 0::numeric) AS juice_ml,
    COALESCE(mc.ml, 0::numeric) AS macerating_ml,
        CASE
            WHEN COALESCE(t.reorder_level, 0::numeric) = 0::numeric THEN 'No level'::text
            WHEN COALESCE(f.qty, 0::numeric) <= 0::numeric THEN 'Out'::text
            WHEN COALESCE(f.qty, 0::numeric) <= t.reorder_level THEN 'Low'::text
            ELSE 'OK'::text
        END AS status,
    'Produce'::text AS action
   FROM perfume_sizes ps
     JOIN perfumes pf ON pf.id = ps.perfume_id
     LEFT JOIN layer_sums f ON f.stock_kind = 'finished'::text AND f.key_id = ps.id
     LEFT JOIN layer_sums b ON b.stock_kind = 'bottled'::text AND b.key_id = ps.id
     LEFT JOIN layer_sums j ON j.stock_kind = 'juice'::text AND j.key_id = pf.id
     LEFT JOIN macerating mc ON mc.perfume_id = pf.id
     LEFT JOIN stock_thresholds t ON t.item_kind = 'size'::text AND t.item_id = ps.id
  WHERE ps.is_active AND (pf.status = ANY (ARRAY['Active Stock'::text, 'In Production'::text]));

-- 3. A PreMix goes straight to Finished Goods after bottling -----------------------
-- Internal only: called by record_bottling and approve_bottling_request. It does nothing for
-- sizes that are not PreMixes.
create or replace function public.finish_premix_internal(p_size uuid, p_date date, p_run uuid)
returns uuid
language plpgsql
security definer
set search_path to 'public'
as $function$
declare
  v_id uuid := gen_random_uuid();
  v_no bigint;
  v_sz public.perfume_sizes;
  v_pf public.perfumes;
  v_qty int;
  v_cost numeric(14,2);
  v_je uuid;
  v_label text;
  v_date date := coalesce(p_date, current_date);
begin
  select * into v_sz from public.perfume_sizes where id = p_size;
  if not found or not v_sz.is_premix then return null; end if;
  select * into v_pf from public.perfumes where id = v_sz.perfume_id;

  -- the bottled units this bottling run just created
  select coalesce(sum(qty_in), 0)::int into v_qty
    from public.stock_layers
   where source = 'bottling' and source_id = p_run and stock_kind = 'bottled';
  if v_qty <= 0 then raise exception 'The bottled stock for this PreMix run was not found.'; end if;

  v_no := nextval('public.packaging_no_seq');
  v_label := 'PK-' || lpad(v_no::text, 4, '0');

  v_cost := public.stock_out('bottled', v_sz.id, v_qty, v_date, 'packaging', v_id);
  if v_cost is null or v_cost <= 0 then
    raise exception 'The bottled PreMix has no recorded cost, so nothing can be posted.';
  end if;

  v_je := public.post_module_entry(v_date, v_label,
            'PreMix ' || v_label || ' - ' || v_qty || ' x ' || v_pf.name || ' ' || v_sz.size_ml || ' ml (straight to finished goods)',
            jsonb_build_array(
              jsonb_build_object('account_code', '1330', 'debit', v_cost, 'memo', v_label || ' ' || v_pf.name || ' ' || v_sz.size_ml || ' ml'),
              jsonb_build_object('account_code', '1320', 'credit', v_cost, 'memo', 'Bottled perfume used')),
            'packaging', v_id);

  insert into public.packaging_runs
    (id, run_no, run_date, perfume_id, size_id, qty, spoiled_qty, bottled_cost,
     materials_cost, spoiled_cost, finished_cost, notes, journal_entry_id)
  values
    (v_id, v_no, v_date, v_pf.id, v_sz.id, v_qty, 0, v_cost,
     0, 0, v_cost, 'PreMix: finished goods straight after bottling', v_je);

  perform public.stock_in('finished', null, null, v_sz.id, v_date, v_qty, v_cost, 'packaging', v_id);
  return v_id;
end;
$function$;

revoke all on function public.finish_premix_internal(uuid, date, uuid) from public, anon, authenticated;

-- 4. record_bottling: a PreMix is finished right after it is bottled -----------------
create or replace function public.record_bottling(p_date date, p_size uuid, p_qty integer, p_bottle uuid, p_sticker uuid, p_spoiled integer, p_notes text, p_actual_ml numeric DEFAULT NULL::numeric, p_confirm boolean DEFAULT false)
 RETURNS jsonb
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
declare
  v_sz public.perfume_sizes;
  v_std numeric;
  v_avail numeric;
  v_ml numeric;
  v_var numeric;
  v_thr numeric;
  v_run uuid;
  v_req uuid;
begin
  if not public.is_active_user() then raise exception 'Not allowed.'; end if;
  if p_qty is null or p_qty <= 0 then raise exception 'Enter how many bottles were filled.'; end if;
  if p_bottle is null or p_sticker is null then raise exception 'Choose the bottle and the bottle sticker.'; end if;
  if p_actual_ml is not null and p_actual_ml <= 0 then raise exception 'The juice used must be more than zero.'; end if;

  select * into v_sz from public.perfume_sizes where id = p_size and is_active;
  if not found then raise exception 'Choose a perfume size.'; end if;

  v_std := p_qty * v_sz.size_ml;
  select coalesce(sum(qty_remaining), 0) into v_avail
    from public.stock_layers where stock_kind = 'juice' and perfume_id = v_sz.perfume_id;
  if p_actual_ml is not null then
    v_ml := p_actual_ml;
  elsif v_std > v_avail and v_avail > 0 then
    v_ml := v_avail;
  else
    v_ml := v_std;
  end if;
  v_var := round(abs(v_ml - v_std) * 100 / v_std, 2);
  select coalesce(bottling_variance_pct, 15) into v_thr from public.business_settings where id;
  v_thr := coalesce(v_thr, 15);

  if v_var > v_thr then
    if public.is_owner() then
      if not coalesce(p_confirm, false) then
        raise exception 'The juice used (% ml) is % percent away from the catalogue amount (% ml). Confirm it to continue.', v_ml, v_var, v_std;
      end if;
    else
      insert into public.bottling_requests
        (run_date, size_id, qty, spoiled_qty, bottle_material_id, sticker_material_id,
         actual_juice_ml, std_juice_ml, variance_pct, notes)
      values
        (coalesce(p_date, current_date), p_size, p_qty, coalesce(p_spoiled, 0), p_bottle, p_sticker,
         v_ml, v_std, v_var, nullif(trim(coalesce(p_notes, '')), ''))
      returning id into v_req;
      return jsonb_build_object('status', 'Pending', 'id', v_req);
    end if;
  end if;

  v_run := public.post_bottling_internal(p_date, p_size, p_qty, p_bottle, p_sticker, p_spoiled, p_notes, v_ml);
  perform public.finish_premix_internal(p_size, coalesce(p_date, current_date), v_run);
  return jsonb_build_object('status', 'Posted', 'id', v_run);
end;
$function$;

-- 5. approve_bottling_request: same for requests approved by the owner ----------------
create or replace function public.approve_bottling_request(p_id uuid)
 RETURNS uuid
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
declare
  q public.bottling_requests;
  v_run uuid;
begin
  if not public.is_owner() then raise exception 'Only the owner can approve a bottling request.'; end if;
  select * into q from public.bottling_requests where id = p_id for update;
  if not found then raise exception 'Request not found.'; end if;
  if q.status <> 'Pending' then raise exception 'This request was already decided.'; end if;

  v_run := public.post_bottling_internal(q.run_date, q.size_id, q.qty, q.bottle_material_id, q.sticker_material_id,
                                         q.spoiled_qty, q.notes, q.actual_juice_ml);
  perform public.finish_premix_internal(q.size_id, q.run_date, v_run);

  update public.bottling_requests
     set status = 'Approved', run_id = v_run, decided_by = auth.uid(), decided_at = now()
   where id = p_id;
  return v_run;
end;
$function$;

-- 6. record_sale: the price list follows the customer's type ------------------------------
-- Same arguments as before (no new parameter), so the app keeps calling it the same way.
create or replace function public.record_sale(
  p_date date,
  p_customer uuid,
  p_payment text,
  p_notes text,
  p_lines jsonb,
  p_down_payment numeric default 0,
  p_down_method text default null,
  p_down_reference text default null
)
returns uuid
language plpgsql
security definer
set search_path to 'public'
as $function$
declare
  v_id uuid := gen_random_uuid();
  v_no bigint;
  v_today date := (now() at time zone 'Asia/Manila')::date;
  v_date date := coalesce(p_date, (now() at time zone 'Asia/Manila')::date);
  v_line jsonb;
  v_n int := 0;
  v_qty int;
  v_sz public.perfume_sizes;
  v_pname text;
  v_price numeric(14,2);
  v_total numeric(14,2) := 0;
  v_down numeric(14,2) := round(coalesce(p_down_payment, 0), 2);
  v_dmethod text := nullif(trim(coalesce(p_down_method, '')), '');
  v_dref text := nullif(trim(coalesce(p_down_reference, '')), '');
  v_tier text := 'retail';   -- price list: follows the customer's type (retail, reseller or distributor)
begin
  if not public.is_active_user() then raise exception 'Not allowed.'; end if;
  if p_payment is null or p_payment not in ('cash','bank','credit') then
    raise exception 'Choose how the customer pays.';
  end if;
  if p_lines is null or jsonb_typeof(p_lines) <> 'array' or jsonb_array_length(p_lines) < 1 then
    raise exception 'Add at least one item.';
  end if;

  if v_down < 0 then raise exception 'The upfront payment cannot be negative.'; end if;
  if v_down > 0 then
    if p_payment <> 'credit' then
      raise exception 'An upfront payment only applies to credit sales.';
    end if;
    if v_dmethod is null or v_dmethod not in ('cash','bank') then
      raise exception 'Choose whether the upfront payment was cash or bank transfer.';
    end if;
    if v_dmethod = 'bank' and v_dref is null then
      raise exception 'Enter the bank transfer reference number for the upfront payment.';
    end if;
    if v_date > v_today then
      raise exception 'A sale with an upfront payment cannot be dated in the future.';
    end if;
  else
    v_dmethod := null;
    v_dref := null;
  end if;

  if p_customer is not null then
    perform 1 from public.customers where id = p_customer and status = 'Active';
    if not found then raise exception 'Customer not found or inactive.'; end if;
    select lower(coalesce(category, 'Retail')) into v_tier from public.customers where id = p_customer;
  end if;
  if p_payment = 'credit' then
    if p_customer is null then raise exception 'A credit sale needs a customer.'; end if;
    perform 1 from public.customer_credit where customer_id = p_customer and credit_allowed;
    if not found then
      raise exception 'This customer is not allowed credit. Choose cash or bank, or ask the owner to allow credit.';
    end if;
  end if;

  v_no := nextval('public.sale_no_seq');
  insert into public.sales (id, sale_no, sale_date, customer_id, payment_type, status, notes,
                            down_payment, down_method, down_reference, price_tier)
  values (v_id, v_no, v_date, p_customer, p_payment, 'Pending', nullif(trim(coalesce(p_notes, '')), ''),
          v_down, v_dmethod, v_dref, v_tier);

  for v_line in select * from jsonb_array_elements(p_lines) loop
    v_n := v_n + 1;
    v_qty := coalesce((v_line->>'qty')::int, 0);
    if v_qty <= 0 then raise exception 'Line %: quantity must be more than zero.', v_n; end if;

    select ps.* into v_sz
      from public.perfume_sizes ps join public.perfumes pf on pf.id = ps.perfume_id
     where ps.id = nullif(v_line->>'size_id', '')::uuid and ps.is_active and pf.status <> 'Discontinued';
    if not found then raise exception 'Line %: choose a perfume and size.', v_n; end if;
    select name into v_pname from public.perfumes where id = v_sz.perfume_id;
    if v_sz.price <= 0 then
      raise exception 'No selling price is set for % % ml. Set it in the Signature Collection first.', v_pname, trim_scale(v_sz.size_ml);
    end if;

    -- the price list follows the customer's type; a price of 0 on a list means "use the retail price"
    v_price := round(coalesce(nullif(v_line->>'unit_price', '')::numeric,
                              case when v_tier = 'reseller' and v_sz.reseller_price > 0 then v_sz.reseller_price
                                   when v_tier = 'distributor' and v_sz.distributor_price > 0 then v_sz.distributor_price
                                   else v_sz.price end), 2);
    if v_price <= 0 then raise exception 'Line %: the price must be more than zero.', v_n; end if;

    insert into public.sale_lines (sale_id, line_no, size_id, qty, unit_price, list_price, amount)
    values (v_id, v_n, v_sz.id, v_qty, v_price, v_sz.price, round(v_qty * v_price, 2));
    v_total := v_total + round(v_qty * v_price, 2);
  end loop;

  if v_down > 0 and v_down >= v_total then
    raise exception 'The upfront payment must be less than the sale total. For a full payment, use Cash or Bank transfer.';
  end if;

  update public.sales set total = v_total where id = v_id;

  -- cash and bank sales post immediately; credit waits for the owner
  if p_payment <> 'credit' then
    perform public.post_sale_internal(v_id, v_date);
  end if;
  return v_id;
end;
$function$;

revoke all on function public.record_sale(date, uuid, text, text, jsonb, numeric, text, text) from public, anon;
grant execute on function public.record_sale(date, uuid, text, text, jsonb, numeric, text, text) to authenticated;

commit;

-- make the API pick up the new columns
notify pgrst, 'reload schema';
