-- 38_affiliates.sql
-- Influencer codes and commissions. Run AFTER 37_order_requests.sql.
-- Commission = pieces x the influencer's fixed rate per piece, earned when a request is
-- BOTH Paid and Delivered (status 'Fulfilled'). Voided if it stops being both.

create table if not exists public.affiliates (
  id             uuid primary key default gen_random_uuid(),
  code           text not null unique check (code ~ '^[A-Za-z0-9_-]{1,40}$'),
  name           text not null,
  contact        text,
  rate_per_piece numeric not null default 0 check (rate_per_piece >= 0),
  active         boolean not null default true,
  created_at     timestamptz not null default now()
);

create table if not exists public.affiliate_commissions (
  id            uuid primary key default gen_random_uuid(),
  request_id    uuid not null unique references public.order_requests(id) on delete cascade,
  affiliate_id  uuid not null references public.affiliates(id),
  pieces        integer not null,
  rate          numeric not null,
  amount        numeric not null,
  status        text not null default 'Earned' check (status in ('Earned','Void')),
  created_at    timestamptz not null default now(),
  updated_at    timestamptz not null default now()
);

create table if not exists public.affiliate_payouts (
  id            uuid primary key default gen_random_uuid(),
  affiliate_id  uuid not null references public.affiliates(id),
  amount        numeric not null check (amount > 0),
  paid_on       date not null default current_date,
  notes         text,
  created_at    timestamptz not null default now()
);

alter table public.order_requests
  add column if not exists affiliate_id uuid references public.affiliates(id);

-- Security: the public cannot read these tables. Staff (logged in) can.
-- NOTE: this lets every logged-in staff user see affiliates. To make it owner-only,
-- replace "using (true)" with your own owner check.
alter table public.affiliates            enable row level security;
alter table public.affiliate_commissions enable row level security;
alter table public.affiliate_payouts     enable row level security;
do $$
declare t text;
begin
  foreach t in array array['affiliates','affiliate_commissions','affiliate_payouts'] loop
    execute format('drop policy if exists staff_all on public.%I', t);
    execute format('create policy staff_all on public.%I for all to authenticated using (true) with check (true)', t);
  end loop;
end $$;

-- A new request: match its code to an ACTIVE influencer. Unknown/inactive codes are ignored.
create or replace function public.cs_resolve_affiliate() returns trigger
language plpgsql security definer set search_path = public as $$
begin
  if new.ref_code is not null then
    select id into new.affiliate_id from affiliates
    where lower(code) = lower(new.ref_code) and active limit 1;
  end if;
  return new;
end $$;

drop trigger if exists cs_resolve_affiliate on public.order_requests;
create trigger cs_resolve_affiliate before insert on public.order_requests
for each row execute function public.cs_resolve_affiliate();

-- Earn or void the commission whenever status or paid changes.
create or replace function public.cs_sync_commission() returns trigger
language plpgsql security definer set search_path = public as $$
declare
  v_pieces int;
  v_rate   numeric;
begin
  if new.affiliate_id is null then return new; end if;

  if new.status = 'Fulfilled' and new.paid then
    select coalesce(sum(qty), 0) into v_pieces from order_request_lines where request_id = new.id;
    -- keep the rate from when it was first earned; otherwise use the current rate
    select rate into v_rate from affiliate_commissions where request_id = new.id;
    if v_rate is null then
      select rate_per_piece into v_rate from affiliates where id = new.affiliate_id;
    end if;
    insert into affiliate_commissions (request_id, affiliate_id, pieces, rate, amount, status)
    values (new.id, new.affiliate_id, v_pieces, v_rate, v_pieces * v_rate, 'Earned')
    on conflict (request_id) do update
      set pieces = excluded.pieces, rate = excluded.rate, amount = excluded.amount,
          status = 'Earned', updated_at = now();
  else
    update affiliate_commissions set status = 'Void', updated_at = now()
    where request_id = new.id and status <> 'Void';
  end if;
  return new;
end $$;

drop trigger if exists cs_sync_commission on public.order_requests;
create trigger cs_sync_commission after update of status, paid on public.order_requests
for each row execute function public.cs_sync_commission();
