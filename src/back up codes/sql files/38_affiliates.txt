-- =====================================================================
-- Chinito Scento - influencer affiliates: commissions are AUTOMATIC.
-- Run AFTER 37_order_requests.sql. Do NOT run 26_order_requests.sql.
-- Safe to re-run. Run in the Supabase SQL Editor.
--
-- Nothing in order_requests, order_request_lines or the website function is changed.
-- The influencer is read from order_requests.ref_code, which the website already fills in.
--
-- How it works, with no extra clicks
--   * Each influencer has a code, e.g. MARIA, and a fixed commission per piece.
--     Their link is  yoursite/?ref=MARIA
--   * When staff set an order request to Fulfilled AND Paid, the database itself creates the
--     commission: pieces x the influencer's rate at that moment (kept on the record, so a later
--     rate change never alters what was already earned). Only an Active influencer earns.
--   * If the request stops being Fulfilled and Paid (cancelled, marked unpaid), the commission is
--     voided automatically. If it was already paid out, the change is refused with a clear message:
--     void that payout first.
--   * The owner pays an influencer by selecting their unpaid commissions: one payout settles them.
--
-- Owner only: influencers, commissions and payouts. Not posted to the books in this version.
-- =====================================================================

-- Remove helpers from an earlier draft of this file, if they were ever created
drop function if exists public.update_order_request(uuid, text, boolean);
drop function if exists public.sync_affiliate_commission(uuid);

-- ---------- 1. Tables --------------------------------------------------
create table if not exists public.affiliates (
  id uuid primary key default gen_random_uuid(),
  code text not null unique check (code ~ '^[A-Z0-9]{3,20}$'),
  name text not null,
  contact text,
  rate_per_piece numeric(14,2) not null check (rate_per_piece >= 0),
  status text not null default 'Active' check (status in ('Active','Inactive')),
  notes text,
  created_at timestamptz not null default now()
);

create table if not exists public.affiliate_payouts (
  id uuid primary key default gen_random_uuid(),
  payout_no bigint generated always as identity,
  affiliate_id uuid not null references public.affiliates(id),
  pay_date date not null,
  method text not null check (method in ('cash','bank','gcash','other')),
  reference text,
  amount numeric(14,2) not null check (amount > 0),
  notes text,
  status text not null default 'Posted' check (status in ('Posted','Voided')),
  void_reason text,
  voided_at timestamptz,
  voided_by uuid references auth.users(id),
  created_at timestamptz not null default now(),
  created_by uuid references auth.users(id)
);

create table if not exists public.affiliate_commissions (
  id uuid primary key default gen_random_uuid(),
  affiliate_id uuid not null references public.affiliates(id),
  request_id uuid not null unique references public.order_requests(id),
  pieces int not null,
  rate_per_piece numeric(14,2) not null,
  amount numeric(14,2) not null,
  earned_at timestamptz not null default now(),
  status text not null default 'Earned' check (status in ('Earned','Voided')),
  payout_id uuid references public.affiliate_payouts(id),
  void_reason text
);

create index if not exists affiliate_commissions_affiliate_idx on public.affiliate_commissions (affiliate_id);

-- ---------- 2. Row level security: owner only ---------------------------
alter table public.affiliates enable row level security;
alter table public.affiliate_commissions enable row level security;
alter table public.affiliate_payouts enable row level security;

drop policy if exists affiliates_owner_all on public.affiliates;
create policy affiliates_owner_all on public.affiliates
  for all to authenticated using (public.is_owner()) with check (public.is_owner());

drop policy if exists affiliate_commissions_owner_select on public.affiliate_commissions;
create policy affiliate_commissions_owner_select on public.affiliate_commissions
  for select to authenticated using (public.is_owner());

drop policy if exists affiliate_payouts_owner_select on public.affiliate_payouts;
create policy affiliate_payouts_owner_select on public.affiliate_payouts
  for select to authenticated using (public.is_owner());

revoke all on public.affiliates, public.affiliate_commissions, public.affiliate_payouts from anon;

-- ---------- 3. Earn / void the commission automatically -----------------
create or replace function public.cs_sync_commission() returns trigger
language plpgsql security definer set search_path = public as $$
declare
  a public.affiliates;
  c public.affiliate_commissions;
  v_pieces int;
begin
  select * into c from public.affiliate_commissions where request_id = new.id for update;

  if new.status = 'Fulfilled' and new.paid then
    if c.id is null or c.status = 'Voided' then
      select * into a from public.affiliates
       where code = upper(trim(coalesce(new.ref_code, ''))) and status = 'Active';
      if a.id is null then return new; end if;   -- no active influencer on this order
      select coalesce(sum(qty), 0) into v_pieces from public.order_request_lines where request_id = new.id;
      if c.id is null then
        insert into public.affiliate_commissions (affiliate_id, request_id, pieces, rate_per_piece, amount)
        values (a.id, new.id, v_pieces, a.rate_per_piece, round(v_pieces * a.rate_per_piece, 2));
      else
        update public.affiliate_commissions
           set status = 'Earned', affiliate_id = a.id, pieces = v_pieces, rate_per_piece = a.rate_per_piece,
               amount = round(v_pieces * a.rate_per_piece, 2), earned_at = now(), void_reason = null
         where id = c.id;
      end if;
    end if;
  elsif c.id is not null and c.status = 'Earned' then
    if c.payout_id is not null then
      raise exception 'The influencer commission for this order was already paid out. Void that payout first.';
    end if;
    update public.affiliate_commissions
       set status = 'Voided', void_reason = 'The order is no longer both Fulfilled and Paid.'
     where id = c.id;
  end if;
  return new;
end $$;

revoke all on function public.cs_sync_commission() from public, anon, authenticated;

drop trigger if exists cs_sync_commission on public.order_requests;
create trigger cs_sync_commission after update of status, paid on public.order_requests
for each row execute function public.cs_sync_commission();

-- ---------- 4. Pay an influencer (owner only) ---------------------------
create or replace function public.record_affiliate_payout(
  p_affiliate uuid, p_date date, p_method text, p_reference text, p_notes text, p_commission_ids uuid[]
) returns bigint
language plpgsql security definer set search_path = public as $$
declare
  v_id uuid;
  v_no bigint;
  v_n int;
  v_total numeric(14,2);
  v_want int;
begin
  if not public.is_owner() then raise exception 'Only the owner can record a payout.'; end if;
  if p_commission_ids is null or cardinality(p_commission_ids) = 0 then raise exception 'Select at least one commission to pay.'; end if;
  if p_method not in ('cash','bank','gcash','other') then raise exception 'Choose how it was paid.'; end if;

  select count(distinct x) into v_want from unnest(p_commission_ids) x;
  perform 1 from public.affiliate_commissions where id = any(p_commission_ids) for update;
  select count(*), coalesce(sum(amount), 0) into v_n, v_total
    from public.affiliate_commissions
   where id = any(p_commission_ids) and affiliate_id = p_affiliate and status = 'Earned' and payout_id is null;
  if v_n <> v_want then raise exception 'Some of the selected commissions are no longer available. Please reload and try again.'; end if;
  if v_total <= 0 then raise exception 'The payout amount must be more than zero.'; end if;

  insert into public.affiliate_payouts (affiliate_id, pay_date, method, reference, amount, notes, created_by)
  values (p_affiliate, coalesce(p_date, current_date), p_method, nullif(trim(coalesce(p_reference, '')), ''),
          v_total, nullif(trim(coalesce(p_notes, '')), ''), auth.uid())
  returning id, payout_no into v_id, v_no;

  update public.affiliate_commissions set payout_id = v_id where id = any(p_commission_ids);
  return v_no;
end;
$$;

create or replace function public.void_affiliate_payout(p_id uuid, p_reason text)
returns void
language plpgsql security definer set search_path = public as $$
declare
  r public.affiliate_payouts;
begin
  if not public.is_owner() then raise exception 'Only the owner can void a payout.'; end if;
  if coalesce(trim(p_reason), '') = '' then raise exception 'Please give a reason.'; end if;

  select * into r from public.affiliate_payouts where id = p_id for update;
  if not found then raise exception 'Payout not found.'; end if;
  if r.status <> 'Posted' then raise exception 'This payout is already voided.'; end if;

  update public.affiliate_commissions set payout_id = null where payout_id = p_id;
  update public.affiliate_payouts
     set status = 'Voided', void_reason = trim(p_reason), voided_at = now(), voided_by = auth.uid()
   where id = p_id;
end;
$$;

-- ---------- 5. Totals per influencer (owner only) -----------------------
create or replace function public.affiliate_summary()
returns table (affiliate_id uuid, orders_pending int, orders_earned int, pieces int, earned numeric, paid numeric, balance numeric)
language sql stable security definer set search_path = public as $$
  select a.id,
         (select count(*) from public.order_requests r
           where upper(trim(coalesce(r.ref_code, ''))) = a.code and r.status <> 'Cancelled'
             and not exists (select 1 from public.affiliate_commissions c where c.request_id = r.id and c.status = 'Earned'))::int,
         (select count(*) from public.affiliate_commissions c where c.affiliate_id = a.id and c.status = 'Earned')::int,
         coalesce((select sum(c.pieces) from public.affiliate_commissions c where c.affiliate_id = a.id and c.status = 'Earned'), 0)::int,
         coalesce((select sum(c.amount) from public.affiliate_commissions c where c.affiliate_id = a.id and c.status = 'Earned'), 0)::numeric,
         coalesce((select sum(c.amount) from public.affiliate_commissions c where c.affiliate_id = a.id and c.status = 'Earned' and c.payout_id is not null), 0)::numeric,
         coalesce((select sum(c.amount) from public.affiliate_commissions c where c.affiliate_id = a.id and c.status = 'Earned' and c.payout_id is null), 0)::numeric
  from public.affiliates a
  where public.is_owner()
  order by a.name;
$$;

revoke all on function public.record_affiliate_payout(uuid,date,text,text,text,uuid[]) from public, anon;

grant execute on function public.void_affiliate_payout(uuid,text) to authenticated;
grant execute on function public.affiliate_summary() to authenticated;
